## 第 1 步：一個錯誤數字，被 7 個代理「正確地」執行

## 第 2 步：練習 1 原樣重現——角色卡檢查＋四種防護設定對照

- **標題**：寫出 `bus_sim.py`：驗證 3 張角色卡，再讓 A 無防護、B 去重、C 去重＋熔斷、D 再加灰度各跑 60 步
- **目的**：照練習 1 的規格：極簡 Message Bus（msg_id = `agent_id:timestamp:seq`、15% 重送、接收端最近 500 個 ID 去重）、角色卡用 dict 加 JSON Schema 風格檢查，以及庫存錯誤廣播的四種設定。讓學習者看到：去重只消掉「重複」，擋不住「錯誤本身」；熔斷把損失壓在觸發那一步；灰度先送最低風險的下游，損失歸零。
- **提示（tip）**：先看角色卡那三行：pricing-01 自己宣告能退款，又把退款列為禁止動作——這種矛盾人眼容易漏，交給機器檢查。再看表格：B 的「去重擋下」會跟 A 的「重複處理」一模一樣，但 B 的低於成本成交仍是 940。

```python
import importlib
import sys
from pathlib import Path

out = Path("ch08/interactive")
out.mkdir(parents=True, exist_ok=True)
(out / "bus_result.json").unlink(missing_ok=True)
SRC = """
import json
import random
from collections import deque
from pathlib import Path

OUT = Path(__file__).resolve().parent
SEED = 8
STEPS = 60
BAD_STEP = 10
HUMAN_FIX_STEP = 57
TRUE_STOCK = 50
BAD_STOCK = 1000
DOWNSTREAM = ["pricing", "logistics", "support", "ads", "procurement", "warehouse", "report"]
CANARY = "report"
ORDERS_PER_STEP = 20
REPLIES_PER_STEP = 50
NOTIFY_STEPS = 3
SETTINGS = [("A", "無防護"), ("B", "只開去重"), ("C", "去重+熔斷"), ("D", "去重+熔斷+灰度")]

REQUIRED = [("agent_id", str), ("role", str), ("capabilities", list), ("constraints", dict), ("dependencies", list), ("escalation", str)]

CARDS = [
    {"agent_id": "inventory-01", "role": "庫存代理", "capabilities": ["read_stock", "publish_inventory_update"],
     "constraints": {"forbidden_actions": ["set_price", "dispatch_truck"], "max_change_ratio": 5},
     "dependencies": ["warehouse_db"], "escalation": "oncall-sre"},
    {"agent_id": "pricing-01", "role": "定價代理", "capabilities": ["set_price", "issue_refund"],
     "constraints": {"forbidden_actions": ["issue_refund"], "min_margin": 0.05},
     "dependencies": ["inventory-01"], "escalation": "pricing-owner"},
    {"agent_id": "logistics-01", "role": "物流代理", "capabilities": ["dispatch_truck"],
     "constraints": {"forbidden_actions": ["set_price"]},
     "dependencies": "inventory-01"},
]


def validate_card(card):
    errors = []
    for key, typ in REQUIRED:
        if key not in card:
            errors.append("缺少必填欄位 " + key)
        elif not isinstance(card[key], typ):
            errors.append("欄位 {} 型別應為 {}，實際是 {}".format(key, typ.__name__, type(card[key]).__name__))
    caps = card.get("capabilities")
    cons = card.get("constraints")
    if isinstance(caps, list) and isinstance(cons, dict):
        clash = sorted(set(caps) & set(cons.get("forbidden_actions", [])))
        if clash:
            errors.append("forbidden_actions 與 capabilities 衝突：" + ",".join(clash))
    return errors


def run(setting, dup_prob=0.15, window=500, confirm=1, max_lag=20, seed=SEED):
    rng = random.Random(seed)
    dedup = setting in ("B", "C", "D")
    breaker = setting in ("C", "D")
    canary = setting == "D"
    win = {r: deque() for r in DOWNSTREAM}
    win_set = {r: set() for r in DOWNSTREAM}
    ever = {r: set() for r in DOWNSTREAM}
    latest = {r: (0, TRUE_STOCK) for r in DOWNSTREAM}
    pending = {}
    s = {"setting": setting, "name": dict(SETTINGS)[setting], "published": 0, "duplicates_sent": 0,
         "duplicates_processed": 0, "duplicates_blocked": 0, "bad_trucks": 0, "below_cost": 0,
         "bad_replies": 0, "trip_step": None, "trip_reason": "", "fixed_step": HUMAN_FIX_STEP}
    affected = set()
    last_good = TRUE_STOCK
    abnormal = 0
    paused = False
    fix_step = HUMAN_FIX_STEP
    seq = 0
    for t in range(1, STEPS + 1):
        if paused and t >= fix_step:
            paused = False
        published_now = not paused
        stock = BAD_STOCK if BAD_STEP <= t < fix_step else TRUE_STOCK
        deliveries = []
        jump = False
        if published_now:
            seq += 1
            msg = {"msg_id": "inventory-01:{}:{}".format(t, seq), "type": "inventory_update", "ts": t,
                   "payload": {"sku": "PHONE-X", "stock": stock}}
            s["published"] += 1
            jump = stock > last_good * 5 or stock * 5 < last_good
            targets = [CANARY] if (canary and jump) else list(DOWNSTREAM)
            for r in targets:
                deliveries.append((r, msg))
                if rng.random() < dup_prob:
                    s["duplicates_sent"] += 1
                    pending.setdefault(t + rng.randint(0, max_lag), []).append((r, msg))
        deliveries.extend(pending.pop(t, []))
        for r, msg in deliveries:
            mid = msg["msg_id"]
            is_dup = mid in ever[r]
            ever[r].add(mid)
            if dedup and mid in win_set[r]:
                s["duplicates_blocked"] += 1
                continue
            if is_dup:
                s["duplicates_processed"] += 1
            if dedup:
                win[r].append(mid)
                win_set[r].add(mid)
                if len(win[r]) > window:
                    win_set[r].discard(win[r].popleft())
            seen_stock = msg["payload"]["stock"]
            wrong = seen_stock != TRUE_STOCK
            if wrong:
                affected.add(r)
            if r == "logistics" and (wrong or is_dup):
                s["bad_trucks"] += 1
            if msg["ts"] >= latest[r][0]:
                latest[r] = (msg["ts"], seen_stock)
        if canary and jump:
            got = [m for r, m in deliveries if r == CANARY and m["ts"] == t]
            errors = sum(1 for m in got if abs(m["payload"]["stock"] - TRUE_STOCK) > TRUE_STOCK * 0.5)
            rate = errors / len(got) if got else 0.0
            if rate > 0.2:
                latest[CANARY] = (t + 0.5, last_good)
                paused = True
                fix_step = t + NOTIFY_STEPS
                s["trip_step"] = t
                s["trip_reason"] = "灰度：金絲雀 {} 對照倉儲實盤，錯誤率 {:.0%} > 20%，自動回滾".format(CANARY, rate)
        if latest["pricing"][1] > 500:
            s["below_cost"] += ORDERS_PER_STEP
        if latest["support"][1] > 500:
            s["bad_replies"] += REPLIES_PER_STEP
        if breaker and s["trip_step"] is None and published_now:
            abnormal = abnormal + 1 if jump else 0
            if abnormal >= confirm:
                paused = True
                fix_step = t + NOTIFY_STEPS
                s["trip_step"] = t
                s["trip_reason"] = "熔斷：庫存 {} -> {}（{:.0f} 倍 > 5 倍），暫停 inventory_update、下游回到最後正常值".format(last_good, stock, stock / last_good)
                for r in DOWNSTREAM:
                    latest[r] = (t + 0.5, last_good)
        if published_now and not jump:
            last_good = stock
    s["fixed_step"] = fix_step
    s["affected_agents"] = len(affected)
    s["affected_list"] = sorted(affected)
    return s


def main(write=True):
    print("角色卡檢查（JSON Schema 風格：必填欄位、型別、forbidden_actions 不可出現在 capabilities）")
    cards = []
    for card in CARDS:
        errs = validate_card(card)
        cards.append({"agent_id": card.get("agent_id"), "ok": not errs, "errors": errs})
        print("  {:<13} {}".format(card.get("agent_id"), "PASS" if not errs else "FAIL"))
        for e in errs:
            print("      - " + e)
    print()
    results = [run(k) for k, _ in SETTINGS]
    print("Message Bus 模擬：60 步、7 個下游、網路 15% 機率重送（延遲 0～20 步），第 10 步庫存 50 -> 1000")
    head = "{:<18} | {:>8} | {:>8} | {:>8} | {:>12} | {:>12} | {:>6} | {:>6}"
    print(head.format("設定", "重複處理", "去重擋下", "錯誤派車", "低於成本成交", "錯誤客服回覆", "熔斷步", "受影響"))
    print("-" * 106)
    for r in results:
        trip = "-" if r["trip_step"] is None else r["trip_step"]
        print(head.format(r["setting"] + " " + r["name"], r["duplicates_processed"], r["duplicates_blocked"],
                          r["bad_trucks"], r["below_cost"], r["bad_replies"], trip, r["affected_agents"]))
    print()
    for r in results:
        if r["trip_reason"]:
            print("  {}：第 {} 步 {}；第 {} 步人工修好快取後恢復".format(r["setting"], r["trip_step"], r["trip_reason"], r["fixed_step"]))
        else:
            print("  {}：沒有任何機制停下來，第 {} 步才由人工發現並修好".format(r["setting"], r["fixed_step"]))
    if write:
        path = OUT / "bus_result.json"
        data = {"seed": SEED, "cards": cards, "results": results}
        path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
        print()
        print("已寫入", path)
    return results


if __name__ == "__main__":
    main()
"""
```

## 第 3 步：改參數＋找陷阱——熔斷要等幾步？重送多頻繁？去重窗口夠不夠大？

- **標題**：三組掃描：熔斷確認步數 confirm、網路重送機率、去重時間窗 N
- **目的**：(a) 熔斷每多等一步「確認」，損失就多一步，這是誤報與漏報的取捨；(b) 重送越頻繁，沒有去重的重複派車越多，有去重則永遠是 47；(c) 陷阱：規格說「只保留最近 N 個 ID」，但如果 N 小於最長重送延遲，晚到的重複訊息會從窗口外溜進來——冪等不是「有開就好」。
- **提示（tip）**：看 (c) 表：N = 500 和 50 都是 0 漏網；N 一旦小於重送最長延遲（20 步），漏網就冒出來，N = 1 幾乎等於沒開去重。

```python
def static_check(card):
    errs = []
    cons = card.get("constraints", {})
    clash = sorted(set(card.get("capabilities", [])) & set(cons.get("forbidden_actions", [])))
    if clash:
        errs.append("forbidden_actions 與 capabilities 衝突：" + ",".join(clash))
    fd = cons.get("flash_discount")
    if not isinstance(fd, (int, float)) or not 0 <= fd <= 1:
        errs.append("flash_discount 必須是 0～1 的數字")
    return errs

    d = card["constraints"]["flash_discount"] if order["flash"] else 0.0
    return order["base"] * (1 - d)
```

```python
import random

COST = {"classify": 0.1, "open": 1.0, "closed": 10.0}


def make_tickets(seed=8, n=1000):
    rng = random.Random(seed)
    out = []
    for _ in range(n):
        is_complex = rng.random() < 0.4
        conf = rng.uniform(0.4, 1.0) if is_complex else 1.0
        out.append({"complex": is_complex, "conf": conf, "u": rng.random()})
    return out


def schedule(tickets, threshold=None):
    cost, correct, esc = 0.0, 0, 0
    for t in tickets:
        if threshold is None:
            cost += COST["closed"]
            correct += t["u"] < (0.95 if t["complex"] else 0.97)
            continue
        cost += COST["classify"]
        if not t["complex"]:
            correct += t["u"] < 0.98
            continue
        cost += COST["open"]
        if t["conf"] < threshold:
            esc += 1
            cost += COST["closed"]
            correct += t["u"] < 0.95
        else:
            correct += t["u"] < t["conf"]
    return cost, correct, esc

def auction(mode, rounds=20, slots=10):
    served = {"batch": 0, "critical": 0}
    revenue = 0.0
    for _ in range(rounds):
        reqs = [["batch", 5.0, 1.0] for _ in range(10)] + [["critical", 3.0, 3.0] for _ in range(2)]
        free = slots
        if mode == "保留容量 2 格":
            for r in reqs:
                if r[0] == "critical" and free > slots - 2:
                    served["critical"] += 1
                    revenue += r[1]
                    r[0] = "done"
                    free -= 1
        if mode == "優先權加權 x3":
            key = lambda r: r[1] * r[2]
        else:
            key = lambda r: r[1]
        for r in sorted([r for r in reqs if r[0] != "done"], key=key, reverse=True)[:free]:
            served[r[0]] += 1
```
