## 第 1 步：故障是常態，慢 200 毫秒就會塞成長龍

- **標題**：先自己算——一天壞幾個節點？慢 200 ms 的庫存服務會塞多久？
- **目的**：讓學習者用數字看見本章主角「靜態依賴的脆弱」：大規模下故障天天發生（但書上的量級要自己重算），而一個只慢 200 ms 的節點，只要「服務時間 > 到達間隔」就會讓後面所有訂單排隊，塞得比慢的時間還久。
- **提示（tip）**：書上說「一萬個節點、年故障率 1%，每天二十多個出事」——先照算一次；然後看「只慢 200 ms」怎麼變成兩分半的大塞車。
- **要跑的內嵌 Python**：

```python
import random
random.seed(505)

P_DAY = 0.01 / 365
print("A. 年故障率 1%：一天平均有幾個節點出事？")
print("{:>9} | {:>8} | {:>8} | {:>14}".format("節點數", "每年故障", "每天平均", "一天至少壞一個"))
print("-" * 52)
for n in (100, 1000, 10000, 100000, 1000000):
    print("{:>9} | {:>8.0f} | {:>8.2f} | {:>13.1f}%".format(n, n * 0.01, n * P_DAY, (1 - (1 - P_DAY) ** n) * 100))
print()
print("書上寫「一萬個節點、年故障率 1%，平均每天二十多個出事」，照算只有 0.27 個；")
print("每天 27 個要嘛是一百萬個節點，要嘛是每個節點每年平均壞一次。量級差 100 倍，數字一定要自己重算。")
print()

def days(n, p, d=30):
    return [sum(1 for _ in range(n) if random.random() < p) for _ in range(d)]

for label, p in (("年故障率 1%", P_DAY), ("每年壞一次", 1 / 365)):
    x = days(10000, p)
    print("1 萬節點、{}：30 天共 {} 次，每天最少 {}、最多 {}、平均 {:.2f}，有出事的天數 {}/30".format(label, sum(x), min(x), max(x), sum(x) / 30, sum(1 for v in x if v > 0)))
print()
print("B. 購物節：訂單每 100 ms 一筆，庫存同步平常 20 ms；第 60～120 秒每筆慢 200 ms")
free = 0
rows = []
for i in range(3000):
    a = i * 100
    svc = 220 if 60000 <= a < 120000 else 20
    start = max(a, free)
    free = start + svc
    rows.append((a, start - a))
waiting = [a for a, w in rows if w > 0]
worst = max(rows, key=lambda r: r[1])
print("  等待超過 1 秒的訂單：{} 筆".format(sum(1 for a, w in rows if w > 1000)))
print("  最長等待：{:.1f} 秒（第 {:.1f} 秒到達的那筆）".format(worst[1] / 1000, worst[0] / 1000))
print("  排隊從第 {:.1f} 秒開始，到第 {:.1f} 秒才清空：慢了 60 秒，卻塞了 {:.0f} 秒".format(waiting[0] / 1000, waiting[-1] / 1000, (waiting[-1] - waiting[0]) / 1000))
print("  原因：服務 220 ms > 到達間隔 100 ms，每來一筆隊伍就多 120 ms；恢復後每筆只能消化 80 ms")
```

## 第 2 步：練習 1-A 原樣重現——靜態 DAG vs 動態邊權重

- **標題**：1000 筆訂單，W1 在第 300～600 筆生病：固定走 W1 vs 依邊權重選路
- **目的**：照練習 1-A 的規格寫出 `dynamic_dag_sim.py`，看靜態版在故障期的逾時與 p95 爆掉，動態版靠「1 ÷（平均延遲 ×（1 + 失敗率×5））」的邊權重自動把 W1 流量壓到 5% 探測下限；也看見 W1 恢復後流量回來得很慢。
- **提示（tip）**：動態版不是靠閒置的備援，而是把故障當成新資訊，在本地持續調權重。注意最後一行：W1 好了，流量卻不一定馬上回來。
- **要跑的內嵌 Python**：

```python
import json
import sys
from pathlib import Path

MOD = """# 第 5 章練習 1：靜態 DAG 與動態重排，再比三種協作範式（只用標準庫、離線）
# 直接執行：PYTHONUTF8=1 python ch05/interactive/dynamic_dag_sim.py
import json
import random
import statistics
from collections import deque
from pathlib import Path

SEED = 505
BASE = {"inventory": 20, "payment": 30, "logistics": 25, "invoice": 10}
WAREHOUSES = ("W1", "W2", "W3", "W4")
TIMEOUT = 500


def pct(xs, q):
    s = sorted(xs)
    k = max(0, min(len(s) - 1, int(round(q * len(s))) - 1))
    return s[k]


def inv_call(rng, wh, bad):
    lat = BASE["inventory"] * rng.uniform(0.9, 1.1)
    if wh == "W1" and bad:
        if rng.random() < 0.3:
            return float(TIMEOUT), True
        return lat + 200, False
    return lat, False


def downstream(rng):
    return sum(BASE[k] * rng.uniform(0.9, 1.1) for k in ("payment", "logistics", "invoice"))


def raw_weight(h):
    if not h:
        return 1.0 / BASE["inventory"]
    avg = sum(x for x, f in h) / len(h)
    fr = sum(1 for x, f in h if f) / len(h)
    return 1.0 / (avg * (1 + fr * 5))


def weights(hist, floor):
    w = [raw_weight(hist[k]) for k in WAREHOUSES]
    total = sum(w)
    p = [x / total for x in w]
    lo = min(range(len(p)), key=lambda j: p[j])
    if p[lo] < floor:
        others = sum(p) - p[lo]
        p = [floor if j == lo else p[j] * (1 - floor) / others for j in range(len(p))]
    return p


def run_orders(mode, n=1000, floor=0.05, window=20, seed=SEED, bad_from=300, bad_to=600):
    rng = random.Random(seed)
    hist = {k: deque(maxlen=window) for k in WAREHOUSES}
    lats, picks, bad_flags, w1_prob = [], [], [], []
    timeouts = 0
    for i in range(1, n + 1):
        bad = bad_from <= i <= bad_to
        if mode == "static":
            first = "W1"
            w1_prob.append(1.0)
        else:
            p = weights(hist, floor)
            w1_prob.append(p[0])
            first = rng.choices(WAREHOUSES, weights=p)[0]
        t, failed = inv_call(rng, first, bad)
        hist[first].append((t, failed))
        if failed:
            timeouts += 1
            if mode == "static":
                backup = "W2"
            else:
                backup = max((k for k in WAREHOUSES if k != first), key=lambda k: raw_weight(hist[k]))
            t2, _ = inv_call(rng, backup, bad)
            hist[backup].append((t2, False))
            t += t2
        lats.append(t + downstream(rng))
        picks.append(first)
        bad_flags.append(bad)
    bad_lats = [x for x, b in zip(lats, bad_flags) if b]
    bad_picks = [k for k, b in zip(picks, bad_flags) if b]
    rec = None
    if mode != "static":
        for i in range(bad_to + 1, n + 1):
            if w1_prob[i - 1] >= 0.20:
                rec = i - bad_to
                break
    return {
        "mode": mode, "n": n, "floor": floor, "window": window,
        "mean": round(statistics.mean(lats), 1),
        "p95": round(pct(lats, 0.95), 1),
        "timeouts": timeouts,
        "share": {k: round(picks.count(k) / n, 3) for k in WAREHOUSES},
        "bad_mean": round(statistics.mean(bad_lats), 1),
        "w1_share_bad": round(bad_picks.count("W1") / len(bad_picks), 3),
        "w1_buckets": [round(picks[s:s + 100].count("W1") / len(picks[s:s + 100]), 2) for s in range(0, n, 100)],
        "recover_after": rec,
    }


def pipeline(stages, n_tasks=50, copies=None):
    copies = copies or [1] * len(stages)
    free = [[0.0] * c for c in copies]
    starts, ends = [], []
    for _ in range(n_tasks):
        t = 0.0
        for k, d in enumerate(stages):
            j = min(range(copies[k]), key=lambda x: free[k][x])
            st = max(t, free[k][j])
            if k == 0:
                starts.append(st)
            t = st + d
            free[k][j] = t
        ends.append(t)
    lat = [e - b for b, e in zip(starts, ends)]
    makespan = max(ends)
    slowest = max(d / c for d, c in zip(stages, copies))
    return {
        "stages": stages, "copies": copies,
        "min_latency": sum(stages),
        "avg_latency": round(statistics.mean(lat), 1),
        "max_latency": round(max(lat), 1),
        "makespan": round(makespan, 1),
        "throughput": round(n_tasks / (makespan / 1000), 2),
        "steady_throughput": round(1000 / slowest, 2),
        "bottleneck_stage": [k + 1 for k, (d, c) in enumerate(zip(stages, copies)) if d / c == slowest],
    }


def mapreduce(seed=SEED, n=20, straggler=600.0, factor=3, rerun=80.0):
    rng = random.Random(seed)
    d = [rng.uniform(50, 100) for _ in range(n)]
    slow = rng.randrange(n)
    d[slow] = straggler
    normal = [x for k, x in enumerate(d) if k != slow]
    med = statistics.median(normal)
    limit = factor * med
    spec = [limit + rerun if x > limit else x for x in d]
    return {
        "n_maps": n, "straggler_index": slow + 1,
        "median_ms": round(med, 1), "speculate_at_ms": round(limit, 1),
        "slowest_normal_ms": round(max(normal), 1),
        "reduce_start_plain": round(max(d), 1),
        "reduce_start_speculative": round(max(spec), 1),
        "saved_ms": round(max(d) - max(spec), 1),
    }


def blackboard(seed=SEED, agents=8, steps=200, fields=10, p=0.3, hot=0.0):
    rng = random.Random(seed)
    queues = [deque() for _ in range(fields)]
    conflicts, writes, t = 0, 0, 0
    waits = []
    while t < steps or any(queues):
        if t < steps:
            touched = [0] * fields
            for _ in range(agents):
                if rng.random() < p:
                    f = 0 if rng.random() < hot else rng.randrange(fields)
                    touched[f] += 1
                    queues[f].append(t)
                    writes += 1
            conflicts += sum(1 for c in touched if c >= 2)
        for q in queues:
            if q:
                waits.append(t - q.popleft())
        t += 1
    return {
        "writes": writes,
        "conflicts_nolock": conflicts,
        "lock_waited": sum(1 for w in waits if w > 0),
        "lock_avg_wait": round(statistics.mean(waits), 2),
        "lock_max_wait": max(waits),
        "drain_steps": t,
    }


def paradigms():
    return {
        "pipeline": {
            "各階段 100ms": pipeline([100, 100, 100, 100, 100]),
            "第3階段 300ms": pipeline([100, 100, 300, 100, 100]),
            "300ms 階段開 3 份": pipeline([100, 100, 300, 100, 100], copies=[1, 1, 3, 1, 1]),
        },
        "mapreduce": mapreduce(),
        "blackboard": {"均勻寫入": blackboard(), "七成寫同一欄": blackboard(hot=0.7)},
    }


def main():
    out = Path(__file__).resolve().parent
    res = {"orders": {m: run_orders(m) for m in ("static", "dynamic")}, "paradigms": paradigms()}
    (out / "dag_result.json").write_text(json.dumps(res, ensure_ascii=False, indent=2), encoding="utf-8")
    for m, r in res["orders"].items():
        print(m, "平均", r["mean"], "ms，p95", r["p95"], "ms，逾時", r["timeouts"], "次")
    print("已寫入", out / "dag_result.json")


if __name__ == "__main__":
    main()
"""
```

## 第 3 步：改參數找陷阱——防飢餓下限與歷史窗口

- **標題**：富者愈富 vs 探測成本：沒有下限，W1 好了也回不來
- **目的**：讓學習者看見邊權重的雙面刃：權重只依「最近表現」分流，生過病的節點會長期分不到流量（飢餓）；探測下限越高恢復越快，但故障期要多吃逾時；縮短歷史窗口也能加快恢復。
- **提示（tip）**：本步延長到 1500 筆。「靜態」與「下限 5%（規格）」兩列的逾時數、故障期 W1 份額必須和第 2 步一模一樣——同種子、同序列。

## 第 5 步：局部重排——等效性、波及範圍、可回退

- **標題**：庫存節點 W1 掛了：挑誰頂替？要重跑多少？
- **目的**：讓學習者看懂局部重排的三個判斷：①替代輸入是否「等效」（下游要的欄位都有）、②波及範圍只到故障節點的下游、③在候選副本上試算，不合格就丟掉，當前圖不動（可回退）。也比較局部重排和整張圖重跑的工作量與完成時間。

```python
         ("logistics", "invoice"), ("pay", "invoice"), ("invoice", "notify"), ("inv_W1", "analytics")]
NEEDS = {"logistics": {"sku", "stock", "warehouse"}, "analytics": {"sku", "stock"}}
CANDS = [("inv_cache", 5, {"sku", "stock"}), ("inv_W2", 22, {"sku", "stock", "warehouse"}), ("inv_W3", 90, {"sku", "stock", "warehouse"})]
BUDGET = 120
FAILED = "inv_W1"
DONE = {"order_in", "pay", "fraud"}


def longest(lat, edges):
    memo = {}
    def f(v):
        if v not in memo:
            memo[v] = lat[v] + max((f(a) for a, b in edges if b == v), default=0)
        return memo[v]
    return max(f(v) for v in lat)


def descendants(edges, v):
    seen, stack = set(), [v]
    while stack:
        x = stack.pop()
        for a, b in edges:
            if a == x and b not in seen:
                seen.add(b)
                stack.append(b)
    return seen


def swap(edges, old, new):
    return [(new if a == old else a, new if b == old else b) for a, b in edges]


snapshot = (tuple(EDGES), tuple(sorted(LAT.items())))
blast = descendants(EDGES, FAILED)
consumers = sorted(b for a, b in EDGES if a == FAILED)

print("原圖關鍵路徑：", longest(LAT, EDGES), "ms；延遲預算", BUDGET, "ms")

    missing = sorted(set().union(*(NEEDS[c] - outs for c in consumers)))

    lat2[name] = lat

    crit = longest(lat2, edges2)
    reasons = []
    if missing:

    if crit > BUDGET:
        reasons.append("超過延遲預算")
    ok = not reasons
    if ok and chosen is None:
        chosen = (name, lat2, edges2)
    print(hdr.format(name, lat, ",".join(missing) or "-", crit, "通過" if ok else "拒絕", "；".join(reasons) or "等效且在預算內"))
print()
print("試算都在副本上做，當前圖沒被改動：", "是" if snapshot == (tuple(EDGES), tuple(sorted(LAT.items()))) else "否")
name, lat2, edges2 = chosen
local_nodes = sorted({name} | blast)
local_work = sum(lat2[v] for v in local_nodes)
local_time = longest({v: (0 if v in DONE else lat2[v]) for v in lat2}, edges2)
full_work = sum(lat2.values())
full_time = longest(lat2, edges2)
print("採用：", name)
hdr2 = "{:<10} | {:>8} | {:>12} | {:>12}"
print(hdr2.format("做法", "重跑節點", "重跑工作量ms", "故障到完成ms"))
print("-" * 52)
print(hdr2.format("局部重排", len(local_nodes), local_work, local_time))
print(hdr2.format("整張重跑", len(lat2), full_work, full_time))
print("局部重排少做 {:.0%} 的工作，早 {} ms 完成".format(1 - local_work / full_work, full_time - local_time))
print("陷阱：只看延遲挑最快的 inv_cache，logistics 會拿不到 warehouse 欄位——等效性要先檢查。")
```

```python
        if p is None:

        elif p not in edges:
            edges.append(p)
    for e in as_list(prop.get("remove_edges")):
        p = pair(e)
        if p is not None and p in edges:
            edges.remove(p)
    return nodes, edges, errs


def critical(nodes, edges, lat=None):
    if lat is None:
        lat = nodes
    indeg = {v: 0 for v in nodes}
    for a, b in edges:
        indeg[b] += 1
    queue = sorted(v for v in nodes if indeg[v] == 0)
    order = []
    while queue:
        v = queue.pop(0)
        order.append(v)
        for a, b in edges:
            if a == v:
                indeg[b] -= 1
                if indeg[b] == 0:
                    queue.append(b)
    if len(order) != len(nodes):
        return None, {}
    finish = {}
    for v in order:
        finish[v] = lat[v] + max((finish[a] for a, b in edges if b == v), default=0.0)
    return max(finish.values()), finish


def reach(edges, start):
    seen = {start}
    stack = [start]
    while stack:
        x = stack.pop()
        for a, b in edges:
            if a == x and b not in seen:
                seen.add(b)
                stack.append(b)
    return seen


def validate(nodes, edges, errs):
    known = [e for e in edges if e[0] in nodes and e[1] in nodes]
    unknown = [e for e in edges if e not in known]

    fwd = reach(known, "inventory") if "inventory" in nodes else set()
    unreach = sorted(set(nodes) - fwd)
```
