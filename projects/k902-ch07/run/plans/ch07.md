## 第 3 步　練習 1 之二：照 scale_coordinates 重寫純函式，做座標縮放表

**目的**：理解模型的座標系是「它看到的那張截圖」，不是實體螢幕；縮放只在長寬比相近、且目標比實體小時才做，放大縮小都用 round；並看清楚官方的 out of bounds 是跟實體螢幕比。

**要跑的內嵌 Python**：

```python
import os, sys
sys.path.insert(0, os.getcwd())
from qs_source import grep

CP = "computer-use-demo/computer_use_demo/tools/computer.py"
print("照 computer.py 的 scale_coordinates 重寫（只用標準庫、不匯入官方模組）：")
for no, line in grep(CP, "abs[(]dimension|if dimension.+< self.width|break|x > self.width or y > self.height|return round"):
    print("  第 {} 行：{}".format(no, line.strip()))
print()

TARGETS = [("XGA", 1024, 768), ("WXGA", 1280, 800), ("FWXGA", 1366, 768)]

def target_for(w, h):
    ratio = w / h
    for name, tw, th in TARGETS:
        if abs(tw / th - ratio) < 0.02:
            if tw < w:
                return (name, tw, th)
            return None
    return None

def to_model(w, h, x, y):
    t = target_for(w, h)
    if t is None:
        return x, y
    return round(x * (t[1] / w)), round(y * (t[2] / h))

def to_physical(w, h, x, y):
    t = target_for(w, h)
    if t is None:
        return x, y
    if x > w or y > h:
        raise ValueError("Coordinates {}, {} are out of bounds".format(x, y))
    return round(x / (t[1] / w)), round(y / (t[2] / h))

print("【縮放表】模型座標 (683,384)")
print("{:<10} | {:<34} | {:<16} | {}".format("實體大小", "縮放目標", "模型看到的截圖", "(683,384) → 實體座標"))
print("-" * 100)
for w, h in [(1920, 1080), (1920, 1200), (1024, 768), (4096, 2160)]:
    t = target_for(w, h)
    if t:
        why = t[0] + " " + str(t[1]) + "x" + str(t[2])
    elif any(abs(tw / th - w / h) < 0.02 for _, tw, th in TARGETS):
        why = "不縮放（比例相同，但寬度沒有更小）"
    else:
        why = "不縮放（找不到相近比例）"
    sw, sh = to_model(w, h, w, h)
    print("{:<10} | {:<34} | {:<16} | {}".format(str(w) + "x" + str(h), why, str(sw) + "x" + str(sh), to_physical(w, h, 683, 384)))
print("  註：1920x1200 的 x 算出 1024.5，Python 的 round 取偶數得 1024（官方算式也一樣）。")
print()

print("【和 tests/tools/computer_test.py 對答案】")
checks = [((1920, 1080), (1366, 768), (1920, 1080)), ((1920, 1200), (1280, 800), (1920, 1200)), ((4096, 2160), (4096, 2160), (4096, 2160))]
for (w, h), xy, want in checks:
    got = to_physical(w, h, *xy)
    print("  {}x{} 模型 {} → {}，測試期望 {}：{}".format(w, h, xy, got, want, got == want))
try:
    to_physical(1920, 1080, 2000, 1500)
    print("  1920x1080 模型 (2000, 1500)：沒有報錯（不對）")
except ValueError as exc:
    print("  1920x1080 模型 (2000, 1500) →", exc, "：True")
```
