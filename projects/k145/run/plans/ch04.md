## 第 1 步：生命遊戲——一條規則，三種命運

- **標題**：同一條「2、3 存活，3 誕生」的規則，方塊靜止、長條振盪、滑翔機會走路
- **目的**：讓學習者看到「規則在個體層、模式在群體層」：每個細胞只看 8 個鄰居，卻冒出靜止、振盪、移動三種整體行為，單看規則猜不到。
- **提示（tip）**：規則只有一行，但模式要跑了才看得到——這就是湧現。
- **要跑的內嵌 Python**：

```python
import random
from collections import Counter

def step(cells):
    cnt = Counter((x + dx, y + dy) for x, y in cells for dx in (-1, 0, 1) for dy in (-1, 0, 1) if (dx, dy) != (0, 0))
    return {c for c, k in cnt.items() if k == 3 or (k == 2 and c in cells)}

def shape(cells):
    mx = min(x for x, y in cells)
    my = min(y for x, y in cells)
    return frozenset((x - mx, y - my) for x, y in cells), (mx, my)

PATTERNS = [
    ("方塊 block", {(0, 0), (1, 0), (0, 1), (1, 1)}),
    ("長條 blinker", {(0, 1), (1, 1), (2, 1)}),
    ("滑翔機 glider", {(1, 0), (2, 1), (0, 2), (1, 2), (2, 2)}),
]
print("規則：活細胞有 2 或 3 個活鄰居就存活；死細胞剛好 3 個活鄰居就誕生；其他都死。")
print()
print("{:<14} | {:>8} | {:>4} | {:>12} | {}".format("初始配置", "活細胞數", "週期", "每週期位移", "類型"))
print("-" * 56)
for name, cells in PATTERNS:
    s0, o0 = shape(cells)
    cur = set(cells)
    period, disp = None, (0, 0)
    for g in range(1, 9):
        cur = step(cur)
        if not cur:
            break
        s, o = shape(cur)
        if s == s0:
            period, disp = g, (o[0] - o0[0], o[1] - o0[1])
            break
    kind = "移動" if disp != (0, 0) else ("靜止" if period == 1 else "振盪")
    print("{:<14} | {:>8} | {:>4} | {:>12} | {}".format(name, len(cells), str(period), str(disp), kind))
print()
print("滑翔機第 0～4 代（# 是活細胞）：")
frames = [set(PATTERNS[2][1])]
for _ in range(4):
    frames.append(step(frames[-1]))
for y in range(5):
    print("   ".join("".join("#" if (x, y) in f else "." for x in range(5)) for f in frames))
print()
rng = random.Random(404)
soup = {(x, y) for x in range(10) for y in range(10) if rng.random() < 0.3}
print("隨機 10x10、密度 30% 的「湯」（seed 404），活細胞數隨世代變化：")
cur = set(soup)
line = []
for g in range(0, 51):
    if g % 10 == 0:
        line.append("第{}代={}".format(g, len(cur)))
    cur = step(cur)
print("  " + "，".join(line))
print("同一條規則，換個初始配置，整體命運就完全不同。")
```

- **預期看到什麼**：
  - 表格 3 列，數字完全確定：
    - `方塊 block | 4 | 1 | (0, 0) | 靜止`
    - `長條 blinker | 3 | 2 | (0, 0) | 振盪`
    - `滑翔機 glider | 5 | 4 | (1, 1) | 移動`
  - 5 個並排的 5×5 小格子：第 0 代與第 4 代形狀相同，但第 4 代整體往右下各移 1 格。
  - 最後一行「第0代=…，第10代=…，…，第50代=…」共 6 個數；第 0 代約 20～40（10×10×0.3 附近），之後的數字起伏不規則（照實記錄，不需要特定值）。

## 第 2 步：寫出 Boids 模組，看「整齊度」自己長出來＋雙胞胎模擬

- **標題**：寫出 `emergence_sim.py`；對齊開與不開，群體整齊度怎麼隨時間演變；再用雙胞胎估 Lyapunov
- **目的**：建立練習 1、練習 2 共用的 Boids 模組；讓學習者看到「要量群體指標（polarization）才看得到湧現」，並體會 Lyapunov 是事後指標——跑 10 步和跑 150 步估出來的值不一樣。
- **提示（tip）**：每隻鳥只管三件事：別太近、跟鄰居同方向、往鄰居中心靠。整齊不是被命令出來的。

```python
import math, sys
from pathlib import Path
sys.dont_write_bytecode = True

MOD = """import math
import random

W = 100.0
HALF = 50.0
NEIGHBOR_R = 15.0
SEP_R = 4.0
MAX_SPEED = 2.0


def classify(pol, nn):
    if pol > 0.8 and nn < 8:
        return "定向運動"
    if nn > 12:
        return "分散"
    if pol < 0.3:
        return "無序"
    return "混沌邊緣"


def polarization(vx, vy):
    sx = 0.0
    sy = 0.0
    for a, b in zip(vx, vy):
        s = math.hypot(a, b)
        if s > 0:
            sx += a / s
            sy += b / s
    return math.hypot(sx, sy) / len(vx)


def nearest_mean(xs, ys):
    n = len(xs)
    total = 0.0
    for i in range(n):
        best = 1e9
        for j in range(n):
            if i == j:
                continue
            dx = abs(xs[j] - xs[i])
            dy = abs(ys[j] - ys[i])
            if dx > HALF:
                dx = W - dx
            if dy > HALF:
                dy = W - dy
            d = math.hypot(dx, dy)
            if d < best:
                best = d
        total += best
    return total / n


def run_boids(sep, ali, coh, seed, steps=150, n=40, perturb=0.0, sep_r=SEP_R, record=False):
    rng = random.Random(seed)
    xs = [rng.uniform(0, W) for _ in range(n)]
    ys = [rng.uniform(0, W) for _ in range(n)]
    vx = [rng.uniform(-1, 1) for _ in range(n)]
    vy = [rng.uniform(-1, 1) for _ in range(n)]
    xs[0] = (xs[0] + perturb) % W
    nr2 = NEIGHBOR_R * NEIGHBOR_R
    sr2 = sep_r * sep_r
    collisions = 0
    pol_hist = [polarization(vx, vy)]
    hist = [(list(xs), list(ys))] if record else None
    for _t in range(steps):
        sx = [0.0] * n
        sy = [0.0] * n
        ax = [0.0] * n
        ay = [0.0] * n
        cx = [0.0] * n
        cy = [0.0] * n
        cnt = [0] * n
        for i in range(n):
            xi = xs[i]
            yi = ys[i]
            for j in range(i + 1, n):
                dx = xs[j] - xi
                if dx > HALF:
                    dx -= W
                elif dx < -HALF:
                    dx += W
                dy = ys[j] - yi
                if dy > HALF:
                    dy -= W
                elif dy < -HALF:
                    dy += W
                d2 = dx * dx + dy * dy
                if d2 < 1.0:
                    collisions += 1
                if 0 < d2 < sr2:
                    d = math.sqrt(d2)
                    ux = dx / d
                    uy = dy / d
                    sx[i] -= ux
                    sy[i] -= uy
                    sx[j] += ux
                    sy[j] += uy
                if d2 < nr2:
                    cnt[i] += 1
                    cnt[j] += 1
                    ax[i] += vx[j]
                    ay[i] += vy[j]
                    ax[j] += vx[i]
                    ay[j] += vy[i]
                    cx[i] += dx
                    cy[i] += dy
                    cx[j] -= dx
                    cy[j] -= dy
        nvx = [0.0] * n
        nvy = [0.0] * n
        for i in range(n):
            a = vx[i] + sep * sx[i] + rng.uniform(-0.05, 0.05)
            b = vy[i] + sep * sy[i] + rng.uniform(-0.05, 0.05)
            k = cnt[i]
            if k:
                a += ali * (ax[i] / k - vx[i]) + coh * (cx[i] / k)
                b += ali * (ay[i] / k - vy[i]) + coh * (cy[i] / k)
            s = math.hypot(a, b)
            if s > MAX_SPEED:
                a = a * MAX_SPEED / s
                b = b * MAX_SPEED / s
            nvx[i] = a
            nvy[i] = b
        vx = nvx
        vy = nvy
        for i in range(n):
            xs[i] = (xs[i] + vx[i]) % W
            ys[i] = (ys[i] + vy[i]) % W
        pol_hist.append(polarization(vx, vy))
        if record:
            hist.append((list(xs), list(ys)))
    pol = pol_hist[-1]
    nn = nearest_mean(xs, ys)
    return {"sep": sep, "ali": ali, "coh": coh, "seed": seed, "sep_r": sep_r,
            "polarization": pol, "nn": nn, "collisions": collisions,
            "phase": classify(pol, nn), "pol_hist": pol_hist, "hist": hist}


def twin_distance(h1, h2):
    out = []
    for (x1, y1), (x2, y2) in zip(h1, h2):
        s = 0.0
        for a, b in zip(x1 + y1, x2 + y2):
            d = abs(a - b)
            if d > HALF:
                d = W - d
            s += d * d
        out.append(math.sqrt(s))
    return out


def lyapunov(sep, ali, coh, seed, steps=150, n=40, sep_r=SEP_R, eps=1e-3):
    base = run_boids(sep, ali, coh, seed, steps=steps, n=n, perturb=0.0, sep_r=sep_r, record=True)
    twin = run_boids(sep, ali, coh, seed, steps=steps, n=n, perturb=eps, sep_r=sep_r, record=True)
    d = twin_distance(base["hist"], twin["hist"])
    lam = (math.log(max(d[-1], 1e-12)) - math.log(max(d[0], 1e-12))) / steps
    return base, d, lam
"""
(out / "emergence_sim.py").write_text(MOD, encoding="utf-8")
print("已寫入", out / "emergence_sim.py")
sys.path.insert(0, str(out))
import emergence_sim as es

a = es.run_boids(0.5, 0.0, 0.05, 404)
b = es.run_boids(0.5, 1.0, 0.05, 404)
print()
print("同一個 seed 404、sep=0.5、coh=0.05，只差對齊 ali：")
print("{:>5} | {:>18} | {:>18}".format("步數", "polarization ali=0", "polarization ali=1"))
print("-" * 48)
for t in (0, 10, 25, 50, 100, 150):
    print("{:>5} | {:>18.3f} | {:>18.3f}".format(t, a["pol_hist"][t], b["pol_hist"][t]))
print()
for label, r in (("ali=0.0", a), ("ali=1.0", b)):
    print("{}：最近鄰距離 {:.2f}，累計碰撞 {}，相 = {}".format(label, r["nn"], r["collisions"], r["phase"]))
print()
base, d, lam = es.lyapunov(0.5, 0.3, 0.05, 404)
print("雙胞胎模擬（sep=0.5, ali=0.3, coh=0.05）：第 0 隻鳥初始 x 只差 0.001")

print("-" * 48)
print("{:>5} | {:>12.3e} | {:>22}".format(0, d[0], "-"))
for t in (10, 30, 75, 150):
    est = (math.log(max(d[t], 1e-12)) - math.log(d[0])) / t
    print("{:>5} | {:>12.3e} | {:>22.4f}".format(t, d[t], est))
```

## 第 4 步：陷阱——為什麼「分散」幾乎不出現？改分離半徑看相跳變

- **標題**：門檻 12 比「隨便亂撒」還遠：分離半徑只有 4，推不出分散相
- **目的**：讓學習者學會檢查分類門檻是否可達：40 隻隨機撒在 100×100 的平均最近鄰距離只有約 7.9，分離半徑 4 根本推不到 12；把分離半徑放大，才看得到系統跳到「分散」相。
- **提示（tip）**：相圖畫不出某個相，先懷疑門檻和規則的作用範圍，而不是懷疑世界。
- **要跑的內嵌 Python**：

```python
sys.dont_write_bytecode = True
sys.path.insert(0, "ch04/interactive")
import emergence_sim as es
random.seed(404)
rng = random.Random(404)
vals = []
for _ in range(20):
```

## 第 5 步：練習 1B——螞蟻選路：只有正反饋、照規格加負反饋、力道對等的負反饋

- **目的**：讓學習者看到正反饋負責聚焦、但會偶爾鎖死在次優解；也看到負反饋若力道比正反饋小好幾個量級，加了等於沒加——兩邊要大致平衡才會分流。
- **提示（tip）**：設計規則時同時問：正反饋是什麼？負反饋是什麼？兩邊力量是否在同一個量級？

```python
import json, random
from pathlib import Path
random.seed(404)
out = Path("ch04/interactive")
out.mkdir(parents=True, exist_ok=True)
LEN = {"short": 1.0, "long": 2.0}
MODES = [
    ("A", "只有正反饋"),
    ("B", "加擁擠成本（規格：最近 20 隻超過 15 隻，感知長度 x1.5）"),
    ("C", "B + 擁擠時該路再蒸發 30% + 費洛蒙下限 0.5"),
]

def run(seed, mode):
    rng = random.Random(seed)
    tau = {"short": 1.0, "long": 1.0}
    picks = []
    for _ in range(500):
        recent = picks[-20:]
        crowd = {}
        for p in ("short", "long"):
            crowded = mode != "A" and recent.count(p) > 15
            crowd[p] = 1.5 if crowded else 1.0
        w_s = (tau["short"] / crowd["short"]) ** 2
        w_l = (tau["long"] / crowd["long"]) ** 2
        choice = "short" if rng.random() * (w_s + w_l) < w_s else "long"
        tau[choice] += 1.0 / (LEN[choice] * crowd[choice])
        for p in tau:
            if mode == "C" and crowd[p] > 1.0:
                tau[p] *= 0.7
            tau[p] *= 0.98
            if mode == "C":
                tau[p] = max(tau[p], 0.5)
        picks.append(choice)
    return picks[-100:].count("short") / 100

res = {m: [run(404 + k, m) for k in range(20)] for m, _ in MODES}
print("500 隻螞蟻、短路長 1、長路長 2，選擇機率正比於 (費洛蒙 / 感知長度倍率)^2，每隻後全部蒸發 2%")
print("20 次（seed 404～423，三種版本用同一批 seed）最後 100 隻走短路的比例：")
print()
print("{:>5} | {:>6} | {:>6} | {:>6}".format("seed", "A", "B", "C"))
print("-" * 32)
for k in range(20):
    print("{:>5} | {:>6.2f} | {:>6.2f} | {:>6.2f}".format(404 + k, res["A"][k], res["B"][k], res["C"][k]))
print()
print("{:<4} | {:>6} | {:>6} | {:>6} | {:>12} | {:>10} | {}".format("版本", "平均", "最小", "最大", "鎖死長路次數", "=1.00 次數", "說明"))
print("-" * 96)
summary = {}
for m, desc in MODES:
    v = res[m]
    lock = sum(1 for x in v if x < 0.5)
    full = sum(1 for x in v if x >= 0.995)
    summary[m] = {"desc": desc, "ratios": v, "mean": round(sum(v) / len(v), 4), "min": min(v), "max": max(v), "locked_long": lock, "all_short": full}
    print("{:<4} | {:>6.2f} | {:>6.2f} | {:>6.2f} | {:>12} | {:>10} | {}".format(m, summary[m]["mean"], min(v), max(v), lock, full, desc))
print()
print("B 為什麼幾乎沒用：鎖定後短路費洛蒙穩定在約 33，長路蒸發到接近 0，")
print("  機率比是 (費洛蒙比)^2 約上萬倍，x1.5 的擁擠成本只改了 2.25 倍。")
print("C 讓負反饋與正反饋同量級：擁擠就削弱該路，加上下限讓另一條路永遠有機會。")
path = out / "emergence_result.json"
data = {}
if path.exists():
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except ValueError:
        data = {}
data["ants"] = summary
path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
print("已把 ants 結果合併寫入", path)
```

- **預期看到什麼**：
  - 每 seed 一列共 20 列，三欄 A、B、C；數字都是 0.00～1.00 的兩位小數。
  - A：大部分 seed 是 `1.00`（聚焦短路），少數可能是 `0.00` 附近（鎖死長路）；「鎖死長路次數」預期 0～6 次，照實記錄（若為 0，代表短路每隻加 1、長路只加 0.5 的優勢已足以壓過早期偶然）。
  - B：分布和 A 很接近，大多數仍是 `1.00`，「=1.00 次數」與 A 相差不大，鎖死次數 ≤ A 或相近——對應「負反饋太弱等於沒有」。
