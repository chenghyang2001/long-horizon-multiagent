## 第 1 步：4 則內容的正回饋——點擊率越來越好，價值越來越差

- **標題**：寫出 `drift_sim.py`，讓 A 純參與度、B 群體獎勵、C 錨點三種推薦器各跑 30 輪，再做熱記憶誤用實驗
- **目的**：照練習 1 的規格把第 1 步放大成 50 則內容、1000 名使用者：A 會在點擊率上升時讓奇蹟療法擠進清單；B 把「分享」也當獎勵，因為分享也偏愛煽動內容，同樣被帶走；C 加上「不可接受規則」當煞車，把奇蹟療法壓回去，代價是少一點點擊。最後順便跑熱記憶：門檻 0.6 太鬆，會把產品缺陷套成「給補償就好」。
- **提示（tip）**：先看 A 那三欄：點擊率往上、品質往下、奇蹟%往上，就是「指標全面變好、價值變差」。再看 C 用多少點擊率換回品質。

import csv
import random
from pathlib import Path

OUT = Path(__file__).resolve().parent
SEED = 7
N_ITEMS = 50
N_MIRACLE = 5
PICKS = 10
USERS = 1000
ROUNDS = 30
NAMES = {"A": "純參與度", "B": "群體獎勵", "C": "錨點"}

COMP = "給補償就好"
FIX = "退換貨＋通報品管"
HUMAN = "轉人工調查"
HISTORY = [
    ((0.9, 0.1, 0.8), COMP), ((0.8, 0.2, 0.9), COMP), ((0.7, 0.1, 0.9), COMP), ((1.0, 0.2, 0.6), COMP),
    ((0.9, 0.3, 0.7), COMP), ((0.6, 0.1, 0.8), COMP), ((0.8, 0.1, 0.5), COMP), ((0.9, 0.2, 0.9), COMP),
    ((0.7, 0.3, 0.8), COMP), ((1.0, 0.1, 0.7), COMP), ((0.5, 0.2, 0.9), COMP), ((0.8, 0.3, 0.6), COMP),

]
NEW = [

    ("S2", (0.8, 0.2, 0.8), False), ("D2", (0.8, 0.6, 0.3), True),

    ("S4", (1.0, 0.2, 0.5), False), ("D4", (1.0, 0.5, 0.1), True),
    ("S5", (0.5, 0.1, 0.9), False), ("S6", (0.9, 0.2, 0.6), False),
]


def make_items(seed=SEED):
    rng = random.Random(seed)
    items = []
    for i in range(N_ITEMS):
        miracle = i >= N_ITEMS - N_MIRACLE
        if miracle:
            q, s = rng.uniform(0.0, 0.09), rng.uniform(0.91, 0.99)
        else:
            q, s = rng.uniform(0.2, 1.0), rng.uniform(0.0, 0.5)
        ctr = 0.2 * q + 0.6 * s + rng.gauss(0.0, 0.02)
        items.append({"id": i, "quality": q, "sens": s, "ctr": min(0.95, max(0.01, ctr)), "miracle": miracle})
    return items


def pick(rng, weights, k):
    pool = list(range(len(weights)))
    chosen = []
    for _ in range(k):
        total = sum(weights[j] for j in pool)
        r = rng.random() * total
        acc = 0.0
        pos = len(pool) - 1
        for p, j in enumerate(pool):
            acc += weights[j]
            if acc >= r:
                pos = p
                break
        chosen.append(pool.pop(pos))
    return chosen


def run(strategy, power=8, penalty=0.1, limit=0.1, seed=SEED, rounds=ROUNDS):
    items = make_items(seed)
    rng = random.Random(seed + 100)
    n = len(items)
    score = [0.0] * n
    shown = [0] * n
    mult = [1.0] * n
    per_item = USERS // PICKS
    rows = []
    triggers = 0
    for t in range(1, rounds + 1):
        w = [((score[i] + 30.0) / (shown[i] + 100.0)) ** power * mult[i] for i in range(n)]
        chosen = pick(rng, w, PICKS)
        clicks = 0
        for i in chosen:
            c = sum(1 for _ in range(per_item) if rng.random() < items[i]["ctr"])
            sh = sum(1 for _ in range(c) if rng.random() < 0.6 * items[i]["sens"])
            shown[i] += per_item
            score[i] += c + (0.5 * sh if strategy == "B" else 0.0)
            clicks += c
        low = [i for i in chosen if items[i]["quality"] < 0.2]
        fired = strategy == "C" and len(low) / PICKS > limit
        if fired:
            triggers += 1
            for i in low:
                mult[i] *= penalty
        rows.append({
            "round": t,
            "strategy": strategy,
            "ctr": clicks / (per_item * PICKS),
            "quality": sum(items[i]["quality"] for i in chosen) / PICKS,
            "miracle_share": sum(1 for i in chosen if items[i]["miracle"]) / PICKS,
            "anchor_fired": int(fired),
        })
    return rows, triggers


def window(rows, end, size=5):
    part = [r for r in rows if end - size < r["round"] <= end]
    return {k: sum(r[k] for r in part) / len(part) for k in ("ctr", "quality", "miracle_share")}


def cosine(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    na = sum(x * x for x in a) ** 0.5
    nb = sum(x * x for x in b) ** 0.5
    return dot / (na * nb)


def hot_memory(threshold):
    rows = []
    for name, vec, defect in NEW:
        best = max(HISTORY, key=lambda h: cosine(vec, h[0]))
        sim = cosine(vec, best[0])
        decision = best[1] if sim >= threshold else HUMAN
        truth = FIX if defect else COMP
        rows.append({"id": name, "defect": defect, "sim": sim, "nearest": best[1], "decision": decision,
                     "wrong": defect and decision == COMP, "ok": decision == truth})
    return rows


def main(write=True):
    items = make_items()
    mir = [it["ctr"] for it in items if it["miracle"]]
    nor = [it["ctr"] for it in items if not it["miracle"]]
    print("內容池：45 個一般項目 + 5 個奇蹟療法（quality < 0.1、sensationalism > 0.9）")
    print("  一般項目點擊率 {:.2f}～{:.2f}（平均 {:.2f}）".format(min(nor), max(nor), sum(nor) / len(nor)))
    print("  奇蹟療法點擊率 {:.2f}～{:.2f}".format(min(mir), max(mir)))
    print("每輪抽 10 項，每項曝光給 100 人（共 1000 人）；權重 = 平滑後估計點擊率的 8 次方")
    results = {s: run(s) for s in ("A", "B", "C")}
    print()
    print("{:>5} | {:^22} | {:^22} | {:^22}".format("輪次", "A 純參與度", "B 群體獎勵", "C 錨點"))
    sub = "{:>6} {:>6} {:>7}".format("點擊率", "品質", "奇蹟%")
    print("{:>5} | ".format("") + " | ".join([sub] * 3))
    print("-" * 84)
    for end in range(5, ROUNDS + 1, 5):
        cells = []
        for s in ("A", "B", "C"):
            m = window(results[s][0], end)
            cells.append("{:>6.3f} {:>6.3f} {:>6.1f}%".format(m["ctr"], m["quality"], m["miracle_share"] * 100))
        print("{:>5} | ".format("{}-{}".format(end - 4, end)) + " | ".join(cells))
    a0, a1 = window(results["A"][0], 5), window(results["A"][0], ROUNDS)
    c1 = window(results["C"][0], ROUNDS)
    print()
    print("A 第 1-5 輪 -> 第 26-30 輪：點擊率 {:.3f} -> {:.3f}，品質 {:.3f} -> {:.3f}，奇蹟療法 {:.0f}% -> {:.0f}%".format(
        a0["ctr"], a1["ctr"], a0["quality"], a1["quality"], a0["miracle_share"] * 100, a1["miracle_share"] * 100))
    print("C 錨點觸發 {} 次；第 26-30 輪奇蹟療法 {:.0f}%，品質 {:.3f}，點擊率比 A 低 {:.1f}%".format(

        print()
        print("已寫入", path)
    return results


if __name__ == "__main__":
    main()

## 第 3 步：改參數——正回饋放大多少？錨點煞車要多重？

- **標題**：把「權重 = 估計點擊率的 k 次方」的 k 從 1 調到 16，再把錨點的懲罰倍數從 1.0（等於沒煞車）調到 0.01
- **目的**：讓學習者親手看到兩個旋鈕：k 越大，正回饋越強，越快被奇蹟療法帶走；錨點規則就算「每輪都有觸發」，懲罰倍數是 1.0 時也完全沒效果——規則要真的改變權重才是煞車。這也是一個陷阱：監控看到「錨點觸發了 20 次」不代表有在保護。
- **提示（tip）**：penalty = 1.0 那一列，觸發次數最多，結果卻跟 A 一模一樣——「有規則」不等於「有煞車」。
