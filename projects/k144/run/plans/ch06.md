## 第 1 步

1. **標題**：94 天沒當機，但網路悄悄變成兩個超級節點
2. **目的**：用一張 150 個代理的網路重演開場故事：系統每天都「自己變得更有效率」（直徑變小、平均路徑變短），但其實是越來越多通訊依賴同一兩個節點。讓學習者先看到「單一指標變好，不等於網路變健康」。
3. **tip（給腳本用）**：150 個代理，一開始是環狀＋150 條隨機捷徑（共 300 條連線）。接下來 94 天，每天 3 次「代理把一條連線改接到比較熱門的節點」（被選中的機率和度數成正比，而且只往更熱門的節點搬），每次改接後都檢查全網仍然連通（＝沒當機）。在第 0、15、30、45、60、75、94 天各體檢一次：直徑、平均最短路、前兩大節點的度數，以及「這兩個超級節點一出事，有多少通訊要繞遠路或直接斷線」。random.seed(94)。
4. **要跑的內嵌 Python**：

```python
import random
from collections import deque

random.seed(94)
N = 150
NODES = list(range(N))

def bfs(adj, s, banned=frozenset()):
    dist = {s: 0}
    q = deque([s])
    while q:
        v = q.popleft()
        for w in adj[v]:
            if w not in dist and w not in banned:
                dist[w] = dist[v] + 1
                q.append(w)
    return dist

adj = {i: set() for i in NODES}
for i in NODES:
    j = (i + 1) % N
    adj[i].add(j)
    adj[j].add(i)
added = 0
while added < 150:
    a, b = random.randrange(N), random.randrange(N)
    if a != b and b not in adj[a]:
        adj[a].add(b)
        adj[b].add(a)
        added += 1

def rewire():
    for _ in range(300):
        a = random.randrange(N)
        b = random.choice(sorted(adj[a]))
        c = random.choices(NODES, weights=[len(adj[v]) for v in NODES])[0]
        if c == a or c == b or c in adj[a]:
            continue
        if len(adj[c]) <= len(adj[b]) or len(adj[b]) <= 1:
            continue
        adj[a].discard(b)
        adj[b].discard(a)
        adj[a].add(c)
        adj[c].add(a)
        if len(bfs(adj, a)) == N:
            return True
        adj[a].discard(c)
        adj[c].discard(a)
        adj[a].add(b)
        adj[b].add(a)
    return False

def snapshot():
    hubs = sorted(NODES, key=lambda v: (-len(adj[v]), v))[:2]
    hub_set = frozenset(hubs)
    full = [bfs(adj, s) for s in NODES]
    diam = max(max(d.values()) for d in full)
    avg = sum(sum(d.values()) for d in full) / (N * (N - 1))
    total = N * (N - 1) // 2
    hub_pairs = total - (N - 2) * (N - 3) // 2
    detour, cut = 0, 0
    for s in NODES:
        if s in hub_set:
            continue
        without = bfs(adj, s, hub_set)
        for t in range(s + 1, N):
            if t in hub_set:
                continue
            if t not in without:
                cut += 1
            elif without[t] > full[s][t]:
                detour += 1
    return hubs, diam, avg, (hub_pairs + detour + cut) / total, (hub_pairs + cut) / total

marks = [0, 15, 30, 45, 60, 75, 94]
ok_n, fail_n = 0, 0
rows = []
print("150 個代理：環狀 + 150 條隨機捷徑，共 300 條連線；之後每天 3 次「把連線改接到更熱門的節點」")
print()
print(" 天數 | 直徑 | 平均最短路 | 前兩大節點度數 | 兩個超級節點一出事：受影響通訊 | 其中直接斷線")
print("-" * 92)
for day in range(0, 95):
    if day > 0:
        for _ in range(3):
            if rewire():
                ok_n += 1
            else:
                fail_n += 1
    if day in marks:
        hubs, diam, avg, hit, cut = snapshot()
        d1, d2 = len(adj[hubs[0]]), len(adj[hubs[1]])
        rows.append((day, diam, avg, hit))
        print(f" {day:>4} | {diam:>4} | {avg:>10.2f} | {d1:>6}, {d2:<6} | {hit:>30.1%} | {cut:>12.1%}")
print()
connected = len(bfs(adj, 0)) == N
first, last = rows[0], rows[-1]
print(f"94 天共改接成功 {ok_n} 次（找不到合適改接 {fail_n} 次）；全程保持連通 = {connected}（系統沒當機）")
print(f"直徑 {first[1]} → {last[1]}、平均最短路 {first[2]:.2f} → {last[2]:.2f}：看起來越來越有效率")
print(f"但兩個超級節點一出事會受影響的通訊從 {first[3]:.1%} 漲到 {last[3]:.1%}（{last[3] / first[3]:.1f} 倍）")
print("「效率變高」可能只是大家都在依賴同一兩個節點——指標要看趨勢與組合，不能只看單一數字變好。")
```

## 第 2 步

1. **標題**：練習 1A：體檢三指標（聚類、直徑、介數）偏好連結 vs 均勻隨機
2. **目的**：照練習 1 建兩張節點數、邊數都一樣的網路，一張讓「富者越富」自然長出超級節點，一張完全均勻，再用三個體檢指標比較。看出偏好連結的網路「路比較短，但介數集中在少數轉運站」；也先用星狀圖驗證介數算法沒寫錯。
3. **tip**：random.seed(6)。偏好連結：從 3 個互連節點開始，每個新節點連 2 條邊，連到既有節點的機率和度數成正比（150 個節點、297 條邊）；均勻隨機：一樣 150 點、297 邊，隨機配對（可能有少數孤立點，直徑在各連通元件內算）。介數用 Brandes 演算法，正規化除以 (n−1)(n−2)/2。參考值：聚類 0.35～0.50 健康、超過約 0.65 要小心團體盲思；介數超過 0.30 就該規劃備援。兩張網路寫到 ch06/interactive/networks.json。
4. **要跑的內嵌 Python**：

```python
import json
import os
import random
from collections import deque

OUT = "ch06/interactive"
os.makedirs(OUT, exist_ok=True)
random.seed(6)
N = 150

def ba_graph(n, m):
    adj = {0: {1, 2}, 1: {0, 2}, 2: {0, 1}}
    targets = [0, 0, 1, 1, 2, 2]
    for v in range(3, n):
        chosen = set()
        while len(chosen) < m:
            chosen.add(random.choice(targets))
        adj[v] = set()
        for u in sorted(chosen):
            adj[v].add(u)
            adj[u].add(v)
            targets += [u, v]
    return adj

def uniform_graph(n, m):
    adj = {i: set() for i in range(n)}
    count = 0
    while count < m:
        a, b = random.randrange(n), random.randrange(n)
        if a != b and b not in adj[a]:
            adj[a].add(b)
            adj[b].add(a)
            count += 1
    return adj

def n_edges(adj):
    return sum(len(s) for s in adj.values()) // 2

def bfs(adj, s, alive=None):
    dist = {s: 0}
    q = deque([s])
    while q:
        v = q.popleft()
        for w in adj[v]:
            if w not in dist and (alive is None or w in alive):
                dist[w] = dist[v] + 1
                q.append(w)
    return dist

def components(adj, alive=None):
    alive = set(adj) if alive is None else set(alive)
    seen, sizes = set(), []
    for s in sorted(alive):
        if s in seen:
            continue
        d = bfs(adj, s, alive)
        seen.update(d)
        sizes.append(len(d))
    return sizes

def dist_stats(adj, alive=None):
    alive = set(adj) if alive is None else set(alive)
    diam, tot, cnt = 0, 0, 0
    for s in alive:
        d = bfs(adj, s, alive)
        diam = max(diam, max(d.values()))
        tot += sum(d.values())
        cnt += len(d) - 1
    return diam, (tot / cnt if cnt else 0.0)

def clustering(adj):
    total = 0.0
    for v in adj:
        nb = sorted(adj[v])
        k = len(nb)
        if k < 2:
            continue
        links = sum(1 for i in range(k) for j in range(i + 1, k) if nb[j] in adj[nb[i]])
        total += 2 * links / (k * (k - 1))
    return total / len(adj)

def betweenness(adj):
    nodes = sorted(adj)
    bc = dict.fromkeys(nodes, 0.0)
    for s in nodes:
        stack = []
        pred = {v: [] for v in nodes}
        sigma = dict.fromkeys(nodes, 0)
        sigma[s] = 1
        dist = dict.fromkeys(nodes, -1)
        dist[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            stack.append(v)
            for w in adj[v]:
                if dist[w] < 0:
                    dist[w] = dist[v] + 1
                    q.append(w)
                if dist[w] == dist[v] + 1:
                    sigma[w] += sigma[v]
                    pred[w].append(v)
        delta = dict.fromkeys(nodes, 0.0)
        while stack:
            w = stack.pop()
            for v in pred[w]:
                delta[v] += sigma[v] / sigma[w] * (1 + delta[w])
            if w != s:
                bc[w] += delta[w]
    n = len(nodes)
    return {v: bc[v] / ((n - 1) * (n - 2)) for v in nodes}

star = {0: {1, 2, 3, 4}, 1: {0}, 2: {0}, 3: {0}, 4: {0}}
line = {0: {1}, 1: {0, 2}, 2: {1}}
s1 = betweenness(star)[0]
s2 = betweenness(line)[1]
ok_s = "通過" if abs(s1 - 1) < 1e-9 and abs(s2 - 1) < 1e-9 else "不通過：介數算法有錯"
print(f"介數自我檢查：5 節點星狀圖中心 = {s1:.3f}、3 節點直線中間 = {s2:.3f}（都應為 1.000）→ {ok_s}")
print()

BA = ba_graph(N, 2)
UNI = uniform_graph(N, n_edges(BA))
nets = [("偏好連結", BA), ("均勻隨機", UNI)]
stats = {}
for name, adj in nets:
    comps = components(adj)
    diam, avg = dist_stats(adj)
    bc = betweenness(adj)
    stats[name] = {"edges": n_edges(adj), "comps": len(comps), "lcc": max(comps),
                   "clust": clustering(adj), "diam": diam, "avg": avg,
                   "maxdeg": max(len(s) for s in adj.values()), "bc": bc}

sa, su = stats["偏好連結"], stats["均勻隨機"]
labels = [("節點數", lambda s: str(N)), ("邊數", lambda s: str(s["edges"])),
          ("連通元件數", lambda s: str(s["comps"])), ("最大連通元件", lambda s: str(s["lcc"])),
          ("平均聚類係數", lambda s: format(s["clust"], ".3f")), ("網路直徑", lambda s: str(s["diam"])),
          ("平均最短路", lambda s: format(s["avg"], ".2f")), ("最大度數", lambda s: str(s["maxdeg"])),
          ("介數最大值", lambda s: format(max(s["bc"].values()), ".3f"))]
print(" 指標             |   偏好連結 |   均勻隨機")
print("-" * 46)
for lab, f in labels:
    print(f" {lab:<12} | {f(sa):>10} | {f(su):>10}")
print()
for name, adj in nets:
    bc = stats[name]["bc"]
    top = sorted(adj, key=lambda v: (-bc[v], v))[:5]
    print(f"【{name}】介數前 5 名")
    print("  名次 | 節點 | 度數 | 正規化介數 | 超過 0.30？")
    for i, v in enumerate(top, 1):
        flag = "是 → 該規劃備援" if bc[v] > 0.30 else "-"
        print(f"  {i:>4} | {v:>4} | {len(adj[v]):>4} | {bc[v]:>10.3f} | {flag}")
    mx = bc[top[0]]
    over = [v for v in adj if bc[v] > 0.30]
    if not over:
        print(f"  沒有節點超過 0.30；最大值 {mx:.3f}，是門檻的 {mx / 0.30:.0%}")
    else:
        print(f"  共 {len(over)} 個節點超過 0.30")
    print()
ratio = max(sa["bc"].values()) / max(su["bc"].values())
print(f"偏好連結的最大介數是均勻隨機的 {ratio:.1f} 倍：路比較短，靠的是少數「轉運站」")
print("兩張都是稀疏網路，聚類都遠低於 0.35～0.50 的健康帶；這裡要看的是兩者的相對差異與介數集中度。")
with open(OUT + "/networks.json", "w", encoding="utf-8") as f:
    json.dump({name: {str(v): sorted(adj[v]) for v in sorted(adj)} for name, adj in nets}, f, ensure_ascii=False)
print("已寫入 " + OUT + "/networks.json")
```

## 第 3 步

1. **標題**：練習 1B：移除 10% 節點容錯測試＋連鎖觀察
2. **目的**：同一張網路，隨機壞掉 15 個節點，和「挑最關鍵的 15 個打掉」，結果差很多。順便抓一個陷阱：吞吐如果除以「原始全部節點對」，光是少了 15 個節點本身就會掉 19%，每種測試都會被誤判成失敗，所以要以「存活節點之間」為準。
3. **tip**：用第 2 步同一顆種子重建同樣兩張網路（不讀檔）。三種方式各移除 15 個節點（10%）：a. 隨機移除 20 次取平均；b. 依介數由高到低；c. 依度數由高到低。吞吐 = 存活節點之間仍連通的對數 ÷ 存活節點之間原本連通的對數，下降 < 20% 為 PASS。另外列出「舊定義：÷ 原始全部節點對」，讓大家看到陷阱。連鎖觀察：只拿掉介數第 1 名，看最大連通元件和平均最短路怎麼變。結果寫到 ch06/interactive/network_health.md。

    diam, tot, cnt = 0, 0, 0
    for s in alive:
        d = bfs(adj, s, alive)
        diam = max(diam, max(d.values()))
        tot += sum(d.values())
        cnt += len(d) - 1
    return diam, (tot / cnt if cnt else 0.0)

def betweenness(adj):
    nodes = sorted(adj)
    bc = dict.fromkeys(nodes, 0.0)
    for s in nodes:
        stack = []
        pred = {v: [] for v in nodes}
        sigma = dict.fromkeys(nodes, 0)
        sigma[s] = 1
        dist = dict.fromkeys(nodes, -1)
        dist[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            stack.append(v)
            for w in adj[v]:
                if dist[w] < 0:
                    dist[w] = dist[v] + 1
                    q.append(w)
                if dist[w] == dist[v] + 1:
                    sigma[w] += sigma[v]
                    pred[w].append(v)
        delta = dict.fromkeys(nodes, 0.0)
        while stack:
            w = stack.pop()
            for v in pred[w]:
                delta[v] += sigma[v] / sigma[w] * (1 + delta[w])
            if w != s:
                bc[w] += delta[w]
    n = len(nodes)
    return {v: bc[v] / ((n - 1) * (n - 2)) for v in nodes}

    return now / base_surv, now / base_all, diam

import json
import math
import os

OUT = "ch06/interactive"
os.makedirs(OUT, exist_ok=True)
D, DT, STEPS = 1.0, 0.1, 200

def run(k, sat=False):
    x = [1.0, 0.0, 0.0]
    rows = [(0, list(x))]
    for t in range(1, STEPS + 1):
        prev = [x[2], x[0], x[1]]
        drive = [k * math.tanh(3 * p) / 3 if sat else k * p for p in prev]
        x = [x[i] + DT * (-D * x[i] + drive[i]) for i in range(3)]
        if t % 20 == 0:
            rows.append((t, list(x)))
    return rows

def label(k):
    lam = -D + k
    if lam < 0:
        return lam, "STABLE"
    if lam == 0:
        return lam, "臨界"
    return lam, "UNSTABLE"

result = {"linear": {}, "saturated": {}, "sweep": []}
for k in (0.5, 0.95, 1.2):
    lam, tag = label(k)

2. **目的**：先看協調成本怎麼讓系統非線性跳變：代理數從 130 加到 150 只多 15%，回應時間卻暴增二十多倍，所以要畫負載─響應曲線、設預警水位。再看隔離門檻為什麼要乘上負載係數：固定 5% 在高負載時會大量誤觸，動態門檻則可能在高負載時太晚，兩者之間要取捨。

import os
import random

OUT = "ch06/interactive"

random.seed(66)
BASE = 0.05

print("【負載─響應曲線】使用率 = 0.4 + 0.004 × (N - 1)，回應時間 = 1 / (1 - 使用率)，預警水位 80%")

for n in (10, 30, 50, 70, 90, 110, 130, 140, 150):
    ch = n * (n - 1) // 2
    rho = 0.4 + 0.004 * (n - 1)
    resp = 1 / (1 - rho) if rho < 1 else float("inf")

    fix_s = "隔離" if fix else "不隔離"
