```python

SRC = """
import json
import random
from pathlib import Path

OUT = Path(__file__).resolve().parent
SEED = 10
RUNS = 2000
YEARS = list(range(2024, 2033))

# A. 開源貢獻者：2023 年 200 萬人，年成長率每年從 N(0.35, 0.10) 抽，下限 0
BASE_2023 = 2000000
G_MU = 0.35
G_SD = 0.10
BOOK_2026 = 5000000          # 書中：2026 年接近五百萬
PARETO_ALPHA = 1.2           # 星標長尾的帕雷托指數
N_AGENTS = 100000            # 書中：2026 年十萬個開源代理

# B. 私有雲成本拐點（單位：萬元／年，基準年 2023，t = 年份 - 2023）
SIZES = (("小企業", 20), ("中企業", 200), ("大企業", 2000))
CLOUD_PER_AGENT = 3.0        # 公有雲：每個代理每年 3 萬
CLOUD_DROP = 0.05            # 公有雲每年降價 5%
HW_FIXED = 400.0             # 私有部署硬體固定成本（機房、網路、備援，年攤提）
HW_PER_AGENT = 4.0           # 每個代理的算力硬體（年攤提；2023 年 GPU 一度是建議售價的 2-3 倍）
LIC_PER_AGENT = 3.0          # 每個代理的模型授權
LABOR_FIXED = 10.0           # 維運人力固定部分（不會隨時間下降）
LABOR_PER_AGENT = 0.1        # 每多一個代理多出的人力
HW_MU, HW_SD = 0.25, 0.08    # 硬體每年下降率（0.75^4 約 0.32：2027 年降到 2023 年的三分之一以下）
LIC_MU, LIC_SD = 0.30, 0.10  # 授權每年下降率
NEVER = 9999                 # 到 2032 年仍未交叉

# C. 四路徑交互：成長率 = 基礎成長 + 係數 x 上一年各路徑佔比
PATHS = ("A", "B", "C", "D")
NAMES = {"A": "開源", "B": "企業", "C": "國家", "D": "暗黑"}
BASE_GROWTH = {"A": 0.30, "B": 0.25, "C": 0.10, "D": 0.25}
REG_FROM_D = 0.6             # D 引發監管的強度


def matrix(reg):
    return {
        "A": {"D": -reg},                              # 監管壓抑開源
        "B": {"A": 0.30, "C": -0.20, "D": -reg},       # 開源供原料給企業；國家與企業部分競爭；監管壓抑企業
        "C": {"B": -0.10, "D": 0.5 * reg},             # 企業搶走國家的角色；暗黑推動國家級基礎設施
        "D": {"A": 0.20, "D": -0.5 * reg},             # 開源也供原料給暗黑；監管反過來壓 D 自己
    }


def pct(values, p):
    s = sorted(values)
    return s[int(p * (len(s) - 1))]


def part_a():
    random.seed(SEED)
    runs = []
    for _ in range(RUNS):
        v = BASE_2023
        row = []
        for _y in YEARS:
            g = max(0.0, random.gauss(G_MU, G_SD))
            v = v * (1 + g)
            row.append(v)
        runs.append(row)
    table = []
    for i, y in enumerate(YEARS):
        col = [r[i] for r in runs]
        table.append({"year": y, "p10": pct(col, 0.1), "p50": pct(col, 0.5), "p90": pct(col, 0.9)})
    stars = [random.paretovariate(PARETO_ALPHA) for _ in range(N_AGENTS)]
    stars.sort(reverse=True)
    total = sum(stars)
    top = N_AGENTS // 100
    longtail = {
        "top1_share": sum(stars[:top]) / total,
        "over_1000": sum(1 for s in stars if s > 1000),
        "under_10_share": sum(1 for s in stars if s < 10) / N_AGENTS,
        "max_star": stars[0],
    }
    return {"table": table, "longtail": longtail}


def cost_cloud(n, t):
    return CLOUD_PER_AGENT * n * (1 - CLOUD_DROP) ** t


def cross_year(n, hw_drops, lic_drops):
    hw = HW_FIXED + HW_PER_AGENT * n
    lic = LIC_PER_AGENT * n
    labor = LABOR_FIXED + LABOR_PER_AGENT * n
    for t, y in enumerate(YEARS, start=1):
        hw = hw * (1 - hw_drops[t - 1])
        lic = lic * (1 - lic_drops[t - 1])
        if hw + lic + labor < cost_cloud(n, t):
            return y
    return NEVER


def clip(x):
    return min(0.9, max(0.0, x))


def part_b():
    random.seed(SEED)
    years = {name: [] for name, _n in SIZES}
    for _ in range(RUNS):
        hw_d = [clip(random.gauss(HW_MU, HW_SD)) for _y in YEARS]
        lic_d = [clip(random.gauss(LIC_MU, LIC_SD)) for _y in YEARS]
        for name, n in SIZES:
            years[name].append(cross_year(n, hw_d, lic_d))
    out = {}
    for name, n in SIZES:
        ys = years[name]
        out[name] = {
            "agents": n,
            "p10": pct(ys, 0.1),
            "p50": pct(ys, 0.5),
            "p90": pct(ys, 0.9),
            "never_share": sum(1 for y in ys if y == NEVER) / RUNS,
            "no_noise": cross_year(n, [HW_MU] * len(YEARS), [LIC_MU] * len(YEARS)),
        }
    return out


def part_c(reg):
    m = matrix(reg)
    v = {p: 1.0 for p in PATHS}
    rows = []
    for y in YEARS:
        total = sum(v.values())
        s = {p: v[p] / total for p in PATHS}
        new = {}
        for p in PATHS:
            g = BASE_GROWTH[p] + sum(c * s[q] for q, c in m[p].items())
            new[p] = max(0.05, v[p] * (1 + g))
        v = new
        total = sum(v.values())
        rows.append({"year": y, "share": {p: v[p] / total for p in PATHS}, "momentum": dict(v)})
    return rows


def fmt_year(y):
    return "未交叉" if y == NEVER else str(y)


def print_a(res):
    print("A. 開源貢獻者推估（2000 次蒙地卡羅，單位：萬人）")
    print("{:>6} | {:>8} | {:>8} | {:>8} | {:>8}".format("年份", "P10", "中位數", "P90", "P90/P10"))
    print("-" * 50)
    for r in res["table"]:
        mark = "  <- 書中：接近 500 萬" if r["year"] == 2026 else ""
        print("{:>6} | {:>8.0f} | {:>8.0f} | {:>8.0f} | {:>8.2f}{}".format(r["year"], r["p10"] / 1e4, r["p50"] / 1e4, r["p90"] / 1e4, r["p90"] / r["p10"], mark))
    r26 = [r for r in res["table"] if r["year"] == 2026][0]
    print()
    print("2026 年中位數 {:.0f} 萬 vs 書中 500 萬：差 {:+.1%}".format(r26["p50"] / 1e4, r26["p50"] / BOOK_2026 - 1))
    print("2026 年 P10～P90：{:.0f} 萬～{:.0f} 萬，寬 {:.0f} 萬".format(r26["p10"] / 1e4, r26["p90"] / 1e4, (r26["p90"] - r26["p10"]) / 1e4))
    lt = res["longtail"]
    print()
    print("長尾：10 萬個代理的星標（帕雷托 alpha = 1.2）")
    print("  星標前 1%（1000 個）佔總星標：{:.1%}".format(lt["top1_share"]))
    print("  星標 > 1000 的代理：{} 個".format(lt["over_1000"]))
    print("  星標 < 10 的代理佔比：{:.1%}".format(lt["under_10_share"]))
    print("  最高星標：{:.0f}".format(lt["max_star"]))


def print_b(res):
    print("B. 私有部署開始比公有雲便宜的年份（2000 次蒙地卡羅）")
    print("{:<4} | {:>6} | {:>6} | {:>6} | {:>6} | {:>8} | {:>6}".format("規模", "代理數", "P10", "中位數", "P90", "未交叉比例", "無雜訊"))

    def gap(a, b):
        return "到 2032 仍未交叉，差距超過 {} 年".format(2032 - a) if b == NEVER else "{} 年".format(b - a)

    big = res["大企業"]["p50"]
```

## 第 4 步：練習 1-C——四路徑交互，暗黑動能加倍會牽動誰？

## 第 5 步：陷阱——同一個數字，相反的解讀

- **標題**：CVE 一樣多的兩個世界，和「案例 8 倍、全面部署卻不到兩成」
- **目的**：對應教練要點「同一個數字可以有相反解讀」。兩個世界用完全相同的 CVE 總數，差別只在基礎設施層佔比：一個是成長副作用，一個是品質崩壞。再看 Fortune 500 公開案例：只數案例會判「企業路徑爆發」，分辨試用與全面後就不能判加速。這兩條規則就是第 6、7 步判分的標準答案。

```python
    s2 = min(0.95, max(0.0, 0.20 + 0.07 * t + random.gauss(0, 0.02)))
    j1 = "減速" if s1 > 0.5 else "成長副作用"
    j2 = "減速" if s2 > 0.5 else "成長副作用"
    if j2 == "減速" and first_slow is None:
        first_slow = y
    print("{:>6} | {:>6} | {:>10.1%} | {:>10.1%} | {:>10} | {:>10}".format(y, total, s1, s2, j1, j2))

print("只看 CVE 總數，兩個世界一模一樣；看漏洞性質，結論相反。")
```

```python
def build_prompt(sig):
    fmt = {

        "cross_effects": "四條路徑之間如何互相牽動",
    }
    lines = ["你是情境分析代理，負責追蹤 AI 代理群體在 2024-2032 年的四條情境路徑："]
    for p in "ABCD":

    lines.append("輸出格式：" + json.dumps(fmt, ensure_ascii=False))
    return chr(10).join(lines)


def norm(x):
    s = str(x).strip().lower()
    for k, v in ZH.items():
        if k in s:
            return v
    if "slow" in s or "decel" in s:
        return "slowing"
    if "accel" in s:
        return "accelerating"
    if "flat" in s or "stable" in s or "steady" in s:
        return "flat"
    return s


def grade(reply, answers):
    reply = reply if isinstance(reply, dict) else {}
    paths = reply.get("paths") if isinstance(reply.get("paths"), dict) else {}
    keys = reply.get("key_signal_per_path") if isinstance(reply.get("key_signal_per_path"), dict) else {}
    cross = str(reply.get("cross_effects", "")).lower()
    rows = []
    for p in "ABCD":
        got = norm(paths.get(p, ""))
        rows.append({"path": p, "rule": answers[p], "model": got, "result": "PASS" if got == answers[p] else "FAIL"})
    a_text = str(keys.get("A", "")).lower()
    b_text = str(keys.get("B", "")).lower()
    hit = [p for p, ks in CROSS_KEYS.items() if any(k.lower() in cross for k in ks)]
    checks = {
        "A 有看漏洞性質（基礎設施層）": any(k.lower() in a_text for k in A_KEYS),
        "B 有分辨試用與全面部署": any(k.lower() in b_text for k in B_KEYS),
        "cross_effects 提到至少兩條路徑": len(hit) >= 2,
    }
    return {"rows": rows, "score": sum(1 for r in rows if r["result"] == "PASS"), "checks": checks, "cross_paths": hit}


def print_grade(g):
    print("{:<4} | {:<12} | {:<14} | {}".format("路徑", "規則答案", "回覆判斷", "結果"))
    print("-" * 46)
    for r in g["rows"]:
        print("{:<4} | {:<12} | {:<14} | {}".format(r["path"], r["rule"], r["model"][:14], r["result"]))
    print("總分：{} / 4".format(g["score"]))
    for k, ok in g["checks"].items():
        print("  {}：{}".format(k, "PASS" if ok else "FAIL"))
    print("  cross_effects 提到的路徑：", ",".join(g["cross_paths"]) or "無")
```
