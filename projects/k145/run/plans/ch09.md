## 第 1 步：三分之一的算術，和「只滲透 2 個就夠了」

- **標題**：拜占庭容錯門檻表＋九代理分層投票：滲透幾個節點能拿下一整層？
- **目的**：帶出本章主角「失控三模式」的第一個——共識錯亂。先看數學鐵律：N 個節點最多容忍 (N-1)//3 個叛徒；再看分層架構（每層 2/3 同意）裡，攻擊者只要 2/9（22%，還在 BFT 容忍範圍內）就能拿下整個策略層。極光事件的 3/127 更只有 2.4%。
- **提示（tip）**：三分之一是「整群一起投票」時的上限；一旦分層，每一層都是一個小群體，門檻就被切碎了。
- **要跑的內嵌 Python**：

```python
from itertools import combinations

print("拜占庭容錯的算術：N 個節點要容忍 f 個叛徒，必須 N >= 3f+1；達成共識要 2f+1 票")
print("{:>5} | {:>9} | {:>11} | {:>8}".format("N", "最多容忍 f", "需要票數 2f+1", "f / N"))
print("-" * 46)
for n in (4, 7, 9, 10, 30, 127):
    f = (n - 1) // 3
    print("{:>5} | {:>9} | {:>11} | {:>7.1%}".format(n, f, 2 * f + 1, f / n))
print()

LAYERS = {"收集": ["C1", "C2", "C3"], "策略": ["S1", "S2", "S3"], "風險": ["R1", "R2", "R3"]}
ALL = [a for v in LAYERS.values() for a in v]
print("九代理思想實驗：3 層 x 3 個代理，每層要 2 個同意才推進")
print("{:>6} | {:>7} | {:>8} | {:>18} | {:>18}".format("滲透數", "佔比", "組合數", "能控制至少 1 層", "能控制至少 2 層"))
print("-" * 72)
for k in range(1, 6):
    total = one = two = 0
    for combo in combinations(ALL, k):
        s = set(combo)
        ctrl = sum(1 for v in LAYERS.values() if len(s & set(v)) >= 2)
        total += 1
        one += ctrl >= 1
        two += ctrl >= 2
    print("{:>6} | {:>7.1%} | {:>8} | {:>9} ({:>6.1%}) | {:>9} ({:>6.1%})".format(k, k / 9, total, one, one / total, two, two / total))
print()

bad_set = {"S2", "S3"}
print("書中情境：攻擊者只滲透 S2、S3")
for name, v in LAYERS.items():
    bad = len(bad_set & set(v))
    print("  {}層：被滲透 {} / 3 -> {}".format(name, bad, "這一層的決定由攻擊者說了算" if bad >= 2 else "仍是誠實多數"))
print("只滲透 2 / 9 = {:.1%}（N=9 的 BFT 上限是 f=2），卻拿下整個策略層：下游收到的每一份策略都已被扭曲".format(2 / 9))
print()
print("極光事件（情境推演）：")
print("  3 個惡意節點 / 127 個代理 = {:.1%}，遠低於三分之一，仍操縱了內部投票".format(3 / 127))
print("  每秒 4000 次 x 14 分鐘 = {:,} 筆交易；資金 500 萬 -> 約 1.2 億 = {:.0f} 倍".format(4000 * 14 * 60, 1.2e8 / 5e6))
print("  停止指令得到的回覆：共識未達成")
```

## 第 2 步：練習 1-A 原樣重現——共識錯亂與漸進變節

- **標題**：寫出 `failure_modes.py`，跑 N=30 的投票：惡意比例 0%～50%、簡單多數 vs 2f+1，再看「漸進變節」第幾回合跨過三分之一
- **目的**：照練習 1 的 A 段規格：惡意節點「雙向拉扯」（對前一半誠實節點投 0、後一半投 1）。讓學習者看到：規則 (2) 在惡意比例跨過約三分之一後，從「正確」整片轉成「共識未達成」（安全但停擺——極光事件的停止指令就是這樣被卡住）；簡單多數則照樣給答案，到 40% 以上開始出現錯誤／分裂，50% 幾乎全錯。漸進變節則說明「三分之一」會隨時間被逼近。
- **提示（tip）**：盯著 30%、35%、40% 三列：規則 (2) 的「正確」會從九成多掉到七成多，再直接歸零。模型假設：誠實節點自己判斷正確的機率 0.95；規則 (2) 的門檻用設計容忍上限 f=9，即 19 票。
- **要跑的內嵌 Python**：

```python
import importlib
import sys
from pathlib import Path

out = Path("ch09/interactive")
out.mkdir(parents=True, exist_ok=True)
SRC = """
import json
import random
from pathlib import Path

OUT = Path(__file__).resolve().parent
SEED = 9
N = 30
ROUNDS = 200
HONEST_ACC = 0.95
PCTS = list(range(0, 55, 5))
F_TOL = (N - 1) // 3
QUORUM = 2 * F_TOL + 1


def one_round(rng, f, rule):
    h = N - f
    votes = [1 if rng.random() < HONEST_ACC else 0 for _ in range(h)]
    ones = sum(votes)
    zeros = h - ones
    decided = []
    for i in range(h):
        to_me = 0 if i < h // 2 else 1
        c1 = ones + (f if to_me == 1 else 0)
        c0 = zeros + (f if to_me == 0 else 0)
        if rule == "majority":
            if c1 != c0:
                decided.append(1 if c1 > c0 else 0)
            else:
                decided.append(rng.choice([0, 1]))
        else:
            if c1 >= QUORUM:
                decided.append(1)
            elif c0 >= QUORUM:
                decided.append(0)
            else:
                decided.append(None)
    if 0 in decided:
        return "wrong"
    if all(d == 1 for d in decided):
        return "correct"
    return "stalled"


def consensus_sweep(seed=SEED):
    rng = random.Random(seed)
    rows = []
    for pct in PCTS:
        f = N * pct // 100
        row = {"pct": pct, "f": f, "ratio": round(f / N, 3)}
        for rule in ("majority", "quorum"):
            cnt = {"correct": 0, "wrong": 0, "stalled": 0}
            for _ in range(ROUNDS):
                cnt[one_round(rng, f, rule)] += 1
            for k, v in cnt.items():
                row[rule + "_" + k] = round(v / ROUNDS, 3)
        rows.append(row)
    return rows


def gradual_defection(seed=SEED, p=0.005, trials=20, max_rounds=2000):
    rng = random.Random(seed)
    crosses = []
    traj = []
    for k in range(trials):
        mal = 0
        cross = None
        for r in range(1, max_rounds + 1):
            mal += sum(1 for _ in range(N - mal) if rng.random() < p)
            if k == 0 and r % 20 == 0 and r <= 160:
                traj.append((r, mal))
            if cross is None and 3 * mal >= N:
                cross = r
                if k > 0:
                    break
            if k == 0 and cross is not None and r >= 160:
                break
        crosses.append(cross)
    return {"p": p, "first_cross": crosses[0], "trajectory": traj, "crosses": crosses}


def goal_drift(seed=SEED, agents=10, cases=500, rounds=20, eps=0.1, alpha=0.02, high_ratio=0.2, abandon=0.6):
    rng = random.Random(seed)
    q = []
    for _ in range(agents):
        q.append({"high_fast": 0.0, "high_delay": -1.0, "normal_fast": 0.0, "normal_delay": -1.0})
    switched = [None] * agents
    rows = []
    for rd in range(1, rounds + 1):
        st = {"high": 0, "high_delay": 0, "high_abandon": 0, "sat": 0, "local": 0.0, "cost": 0}
        for _ in range(cases):
            a = rng.randrange(agents)
            kind = "high" if rng.random() < high_ratio else "normal"
            cost = 5 if kind == "high" else 1
            if rng.random() < eps:
                act = rng.choice(["fast", "delay"])
            elif q[a][kind + "_fast"] >= q[a][kind + "_delay"]:
                act = "fast"
            else:
                act = "delay"
            if act == "fast":
                sat, local, paid = 1, 1 - 0.5 * cost, cost
            elif rng.random() < abandon:
                sat, local, paid = -1, -1.0, 0
            else:
                sat, local, paid = 0, 0.0, cost
            key = kind + "_" + act
            q[a][key] += alpha * (local - q[a][key])
            st["sat"] += sat
            st["local"] += local
            st["cost"] += paid
            if kind == "high":
                st["high"] += 1
                if act == "delay":
                    st["high_delay"] += 1
                    if sat == -1:
                        st["high_abandon"] += 1
        for a in range(agents):
            if switched[a] is None and q[a]["high_delay"] > q[a]["high_fast"]:
                switched[a] = rd
        rows.append({
            "round": rd,
            "high": st["high"],
            "high_delay_ratio": round(st["high_delay"] / max(1, st["high"]), 3),
            "high_abandon": st["high_abandon"],
            "satisfaction": round(st["sat"] / cases, 3),
            "local_reward": round(st["local"] / cases, 3),
            "cost": st["cost"],
        })
    return {"rows": rows, "switched_round": switched}


def resource_curve(n_min=2, n_max=60, unit=1.0, c_low=0.02, c_high=0.04, noisy_from=31):
    rows = []
    prev = None
    for n in range(n_min, n_max + 1):
        c = c_high if n >= noisy_from else c_low
        overhead = c * n * (n - 1) / 2
        net = n * unit - overhead
        rows.append({
            "n": n,
            "coef": c,
            "output": n * unit,
            "overhead": round(overhead, 4),
            "net": round(net, 4),
            "marginal": None if prev is None else round(net - prev, 4),
        })
        prev = net
    best = max(rows, key=lambda r: (r["net"], -r["n"]))
    return {"rows": rows, "best_n": best["n"], "best_net": best["net"]}


def run_all():
    return {
        "seed": SEED,
        "A_consensus": consensus_sweep(),
        "A_gradual": gradual_defection(),
        "B_goal_drift": goal_drift(),
        "C_resource": resource_curve(),
    }


def save_result():
    data = run_all()
    path = OUT / "failure_result.json"
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
    return path, data


def print_a(rows, grad):
    print("A. 共識錯亂：N = {}，每格 {} 回合，誠實節點自己判斷正確的機率 {}".format(N, ROUNDS, HONEST_ACC))
    print("   規則 (1) 簡單多數；規則 (2) 要 2f+1 = {} 票（f = {}）才算達成，否則回報「共識未達成」".format(QUORUM, F_TOL))
    print("   惡意節點雙向拉扯：對前一半誠實節點投 0、對後一半投 1")
    print("{:>4} | {:>3} | {:>6} || {:>9} | {:>11} || {:>9} | {:>9} | {:>10}".format("格", "f", "f/N", "(1)正確", "(1)錯誤/分裂", "(2)正確", "(2)錯誤", "(2)未達成"))
    print("-" * 92)
    for r in rows:
        mark = ""
        if r["f"] == F_TOL:
            mark = "  <- 可容忍上限"
        elif r["f"] == F_TOL + 1:
            mark = "  <- 跨過三分之一"
        print("{:>3}% | {:>3} | {:>6.1%} || {:>9.1%} | {:>11.1%} || {:>9.1%} | {:>9.1%} | {:>10.1%}{}".format(
            r["pct"], r["f"], r["ratio"], r["majority_correct"], r["majority_wrong"],
            r["quorum_correct"], r["quorum_wrong"], r["quorum_stalled"], mark))
    print()
    print("漸進變節：每回合每個誠實節點有 {:.1%} 機率轉成惡意（從 0 個惡意開始）".format(grad["p"]))
    print("  第 1 次試驗的軌跡：" + "、".join("第 {} 回合 {} 個".format(r, m) for r, m in grad["trajectory"]))
    print("  第 1 次試驗在第 {} 回合跨過三分之一（惡意 >= 10 / 30）".format(grad["first_cross"]))
    cs = sorted(grad["crosses"])
    print("  {} 次試驗：最早第 {} 回合、中位數第 {} 回合、最晚第 {} 回合".format(len(cs), cs[0], cs[len(cs) // 2], cs[-1]))
    print("  跨過之前規則 (2) 一切正常；三分之一不是固定門檻，而是會隨時間被逼近的風險範圍。")
```

## 第 5 步：陷阱——微幅偏向抓不到，「偏離多數」還冤枉好人

- **標題**：S2、S3 每步只多偏 +0.3，單步告警幾乎不響；用「誰偏離多數」去抓，抓到的是誠實的 S1
- **目的**：對應「滲透者常不是突然作亂，而是持續微幅偏向，偏差被下游當成輸入再放大」和教練要點「要看共識過程本身的健康度」。讓學習者看到三件事：(1) 單步閾值告警，叛徒和好人一樣少；(2) 當同層 2/3 被滲透，「跟多數比」會讓誠實的那一個看起來最可疑；(3) 只有拿外部參考（不靠這一層自己投票）、看整段平均，才抓得到 S2、S3；而整體部位已被推高好幾十。
- **提示（tip）**：兩種情境用同一串雜訊，差別只在 S2、S3 有沒有偏 +0.3，所以部位差就是攻擊造成的。
- **要跑的內嵌 Python**：

```python
import random
import statistics

STEPS, START, BIAS, SD, FEED, LIMIT, ALARM, FLAG = 120, 12, 0.3, 0.5, 0.01, 1.5, 1.5, 0.15
NAMES = ["S1", "S2", "S3"]


def run(attack):
    rng = random.Random(9)
    pos = 0.0
    alarms = {n: 0 for n in NAMES}
    dev_med = {n: [] for n in NAMES}
    dev_ref = {n: [] for n in NAMES}
    rejects = 0
    for t in range(1, STEPS + 1):
        base = FEED * pos
        props = {}
        for n in NAMES:
            b = BIAS if attack and n in ("S2", "S3") and t >= START else 0.0
            props[n] = base + b + rng.gauss(0, SD)
        med = statistics.median(list(props.values()))
        for n in NAMES:
            if abs(props[n] - base) > ALARM:
                alarms[n] += 1
            if t >= START:
                dev_med[n].append(props[n] - med)
                dev_ref[n].append(props[n] - base)
        if abs(med) > LIMIT:
            rejects += 1
        else:
            pos += med
    return {
        "pos": pos,
        "alarms": alarms,
        "dev_med": {n: statistics.mean(v) for n, v in dev_med.items()},
        "dev_ref": {n: statistics.mean(v) for n, v in dev_ref.items()},
        "rejects": rejects,
    }


clean, bad = run(False), run(True)
print("九代理管線：收集 -> 策略 S1～S3（取中位數 = 2/3 同意）-> 風險（單步 |決策| > 1.5 才否決）")
print("S2、S3 從第 12 步起每次只多偏 +{}（雜訊標準差 {}）；下游把自己的部位當趨勢輸入（回饋 {}）".format(BIAS, SD, FEED))
print()
print("{:<8} | {:>16} | {:>12}".format("情境", "第 120 步累積部位", "風險層否決"))
print("-" * 44)
print("{:<8} | {:>16.2f} | {:>12}".format("沒有滲透", clean["pos"], clean["rejects"]))
print("{:<8} | {:>16.2f} | {:>12}".format("S2S3滲透", bad["pos"], bad["rejects"]))
print("攻擊造成的部位差：{:.2f}".format(bad["pos"] - clean["pos"]))
print()
print("{:>4} | {:>10} | {:>10} | {:>14} | {:>14} | {}".format("代理", "單步告警(無)", "單步告警(有)", "對同層中位數", "對外部參考", "外部參考判定"))
print("-" * 86)
for n in NAMES:
    ref = bad["dev_ref"][n]
    print("{:>4} | {:>10} | {:>10} | {:>+14.3f} | {:>+14.3f} | {}".format(n, clean["alarms"][n], bad["alarms"][n], bad["dev_med"][n], ref, "可疑" if ref > FLAG else "正常"))
print()
suspect = max(NAMES, key=lambda n: abs(bad["dev_med"][n]))
caught = [n for n in NAMES if bad["dev_ref"][n] > FLAG]
print("用「偏離多數最遠」抓內鬼：抓到", suspect)
print("用外部參考看整段平均（> {}）抓內鬼：抓到".format(FLAG), caught)
print("教訓：被滲透的就是多數時，跟多數比只會冤枉好人；要用不受這一層投票影響的參考。")
```

```python
    checks.append(("trusted_source = " + answer["trusted_source"], src == answer["trusted_source"], src))
    got = resp.get("compromised_agents")
    if isinstance(got, str):
        got = [got]
    if not isinstance(got, list):
        got = []
    got_set = sorted(set(str(x).strip().upper() for x in got))
    checks.append(("compromised_agents = " + ",".join(answer["compromised_agents"]), got_set == answer["compromised_agents"], ",".join(got_set)))
    try:
        idx = int(resp.get("first_anomaly_index"))
    except (TypeError, ValueError):
        idx = None
    checks.append(("first_anomaly_index = " + str(answer["first_anomaly_index"]), idx == answer["first_anomaly_index"], str(idx)))
    low = str(resp.get("why", "")).lower()
    ext = [k for k in EXT_KW if k in low]
    own = [k for k in SELF_KW if k in low]
    checks.append(("why 點出「日誌不能只靠被監督者自己的共識背書」（軟性檢查）", bool(ext) and bool(own), "外部類：" + ("、".join(ext) or "無") + "；自我背書類：" + ("、".join(own) or "無")))
    return checks
```
