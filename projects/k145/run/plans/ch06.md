def ws_graph(n, k=4, p=0.1, seed=SEED):
    rng = random.Random(seed)
    adj = [set() for _ in range(n)]
    for i in range(n):
        for j in range(1, k // 2 + 1):
            t = (i + j) % n
            adj[i].add(t)
            adj[t].add(i)
    for j in range(1, k // 2 + 1):
        for i in range(n):
            t = (i + j) % n
            if rng.random() < p and t in adj[i]:
                choices = [w for w in range(n) if w != i and w not in adj[i]]
                if choices:
                    w = rng.choice(choices)
                    adj[i].discard(t)
                    adj[t].discard(i)
                    adj[i].add(w)
                    adj[w].add(i)
    return adj


def bfs(adj, s):
    dist = {s: 0}
    q = deque([s])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if v not in dist:
                dist[v] = dist[u] + 1
                q.append(v)
    return dist


def path_stats(adj, sources):
    total, count, ecc = 0, 0, 0
    for s in sources:
        d = bfs(adj, s)
        total += sum(d.values())
        count += len(d) - 1
        ecc = max(ecc, max(d.values()))
    return total / count, ecc


def clustering(adj):
    vals = []
    for u in range(len(adj)):
        nb = sorted(adj[u])
        k = len(nb)
        if k < 2:
            vals.append(0.0)
            continue
        links = sum(1 for i in range(k) for j in range(i + 1, k) if nb[j] in adj[nb[i]])
        vals.append(2.0 * links / (k * (k - 1)))
    return sum(vals) / len(vals)

def ba_graph(n=300, m=2, seed=SEED):
    rng = random.Random(seed)
    adj = [set() for _ in range(n)]
    rep = []
    for a in range(m + 1):
        for b in range(a + 1, m + 1):
            adj[a].add(b)
            adj[b].add(a)
            rep += [a, b]
    for v in range(m + 1, n):
        chosen = set()
        while len(chosen) < m:
            chosen.add(rng.choice(rep))
        for t in sorted(chosen):
            adj[v].add(t)
            adj[t].add(v)
            rep += [v, t]
    return adj


def random_graph(n, edges, seed=SEED):
    rng = random.Random(seed)
    adj = [set() for _ in range(n)]
    made = 0
    while made < edges:
        a, b = rng.randrange(n), rng.randrange(n)
        if a != b and b not in adj[a]:
            adj[a].add(b)
            adj[b].add(a)
            made += 1
    return adj


def edge_count(adj):
    return sum(len(s) for s in adj) // 2


def giant_after(adj, removed):
    seen = set(removed)
    best = 0
    for s in range(len(adj)):
        if s in seen:
            continue
        seen.add(s)
        q = deque([s])
        size = 0
        while q:
            u = q.popleft()
            size += 1
            for v in adj[u]:
                if v not in seen:
                    seen.add(v)
                    q.append(v)
        best = max(best, size)
    return best / len(adj)
