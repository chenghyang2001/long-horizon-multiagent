## 第 1 步：三個角色的契約——錯誤找得到是哪一環

- **標題**：Planner 動腦、Executor 動手、Monitor 長眼：把黑盒拆成三個有契約的角色
- **目的**：讓學習者看到，把一個黑盒模型拆成三個有明確輸入／輸出／工具箱的角色後，出錯時程式能指出是哪一個角色錯，只重做那一段；黑盒只知道「最後報告錯了」，只能全部重做。
- **提示（tip）**：每個角色只交自己的產出、只用自己的工具。出錯時沿著契約一關一關檢查，第一個不合格的就是兇手。
- **要跑的內嵌 Python**：

```python
import random
random.seed(202)
ROLES = {
    "planner": {"in": "任務描述", "out": "DAG（節點＋依賴）", "tools": ["write_dag"]},
    "executor": {"in": "DAG 的單一節點", "out": "節點結果（含 rows 欄位）", "tools": ["run_sql", "call_api", "write_file"]},
    "monitor": {"in": "執行日誌＋節點結果", "out": "警告紀錄", "tools": ["read_logs", "raise_alert"]},
}
print("{:<9} | {:<14} | {:<20} | {}".format("角色", "輸入", "輸出", "工具箱"))
print("-" * 76)
for r, spec in ROLES.items():
    print("{:<9} | {:<14} | {:<20} | {}".format(r, spec["in"], spec["out"], ", ".join(spec["tools"])))
print()
print("工具層：每個角色只能用自己的工具")
for role, tool in (("executor", "run_sql"), ("executor", "write_dag"), ("monitor", "call_api"), ("planner", "write_dag")):
    print("  {:<9} 呼叫 {:<10} -> {}".format(role, tool, "ALLOW" if tool in ROLES[role]["tools"] else "DENY"))
print()

DAG = {"A": [], "B": ["A"], "C": ["B"], "D": ["C"], "E": ["D"], "F": ["B"], "G": ["D", "E", "F"], "H": ["G"]}
REDO = {"planner": 40, "executor": 5, "monitor": 1}

def has_cycle(dag):
    indeg = {n: len(ps) for n, ps in dag.items()}
    q = [n for n in dag if indeg[n] == 0]
    seen = 0
    while q:
        n = q.pop()
        seen += 1
        for c, ps in dag.items():
            if n in ps:
                indeg[c] -= 1
                if indeg[c] == 0:
                    q.append(c)
    return seen < len(dag)

def one_run(fault):
    plan = {n: list(ps) for n, ps in DAG.items()}
    if fault == "planner":
        plan["B"].append("H")
    results = {n: {"node": n, "rows": random.randint(100, 999)} for n in DAG}
    results["C"]["growth"] = {"戶外": 5.0, "家電": 0.12}
    if fault == "executor":
        del results[random.choice(sorted(DAG))]["rows"]
    alerts = [k for k, g in results["C"]["growth"].items() if g > 3.0]
    if fault == "monitor":
        alerts = []
    if has_cycle(plan):
        found = "planner"
    elif any("rows" not in r for r in results.values()):
        found = "executor"
    elif any(g > 3.0 for g in results["C"]["growth"].values()) and not alerts:
        found = "monitor"
    else:
        found = None
    return found

faults = [random.choice(("planner", "executor", "monitor")) for _ in range(20)]
hits = sum(1 for f in faults if one_run(f) == f)
cnt = {r: faults.count(r) for r in REDO}
print("注入 20 次錯誤，各角色被注入次數：", cnt)
print("{:<12} | {:>8} | {:>14}".format("架構", "定位成功", "平均重做(分鐘)"))
print("-" * 42)
print("{:<12} | {:>5}/20 | {:>14.1f}".format("單一黑盒模型", 0, 40.0))
print("{:<12} | {:>5}/20 | {:>14.1f}".format("三角色", hits, sum(REDO[f] for f in faults) / 20))
print()
print("黑盒只看得到「最後報告錯了」，只能 8 個節點（40 分鐘）全部重做。")
```

- **預期看到什麼**：
  - 角色表 3 列：planner／executor／monitor，各自的輸入、輸出、工具箱。
  - 工具層 4 行依序：`executor 呼叫 run_sql -> ALLOW`、`executor 呼叫 write_dag -> DENY`、`monitor 呼叫 call_api -> DENY`、`planner 呼叫 write_dag -> ALLOW`。
  - 「各角色被注入次數」三個數加起來 = 20（每個大約 3～12）。
  - 黑盒：`0/20`、平均重做 `40.0`；三角色：`20/20`，平均重做明顯較小（約 8～25 分鐘，= (40×planner 次數 + 5×executor 次數 + 1×monitor 次數) / 20，可用上一行的次數手算驗證）。
- **是否呼叫模型**：否

## 第 2 步：Planner 的產出是 DAG——可被驗證、形狀決定並行

- **標題**：書中電商 DAG：拓撲排序、關鍵路徑、並行寬度，以及程式抓得到的四種錯
- **目的**：DAG 是 Planner「可被驗證」的輸出：多一個循環、打錯依賴、有產出沒人接，程式都能抓到；而圖的形狀（不是代理數）決定能省多少時間——關鍵路徑是總時間下限。
- **提示（tip）**：E 和 F 互不依賴，所以「可以」並行；但 executor 從 2 個加到 3 個，總時間不會再變短。
- **要跑的內嵌 Python**：

```python
from collections import Counter
DAG = {"A": [], "B": ["A"], "C": ["B"], "D": ["C"], "E": ["D"], "F": ["B"], "G": ["D", "E", "F"], "H": ["G"]}
EST = {n: 5.0 for n in DAG}

def check(dag):
    probs = []
    for n, ps in dag.items():
        for p in ps:
            if p not in dag:
                probs.append("可執行：{} 依賴不存在的 {}".format(n, p))
    if probs:
        return probs, None
    indeg = {n: len(ps) for n, ps in dag.items()}
    kids = {n: [] for n in dag}
    for n, ps in dag.items():
        for p in ps:
            kids[p].append(n)
    q = sorted(n for n in dag if indeg[n] == 0)
    order = []
    while q:
        n = q.pop(0)
        order.append(n)
        for c in kids[n]:
            indeg[c] -= 1
            if indeg[c] == 0:
                q.append(c)
        q.sort()
    if len(order) < len(dag):
        probs.append("可執行：有循環，卡住的節點 " + ",".join(sorted(set(dag) - set(order))))
        return probs, None
    sinks = sorted(n for n in dag if not kids[n])
    if len(sinks) > 1:
        probs.append("完整：終點不只一個（" + ",".join(sinks) + "），有中間產出沒人接")
    return probs, order

cases = [
    ("原圖", DAG),
    ("B 多依賴 H（循環）", dict(DAG, B=["A", "H"])),
    ("G 的依賴打錯成 E2", dict(DAG, G=["D", "E2", "F"])),
    ("拿掉 G 對 E 的依賴", dict(DAG, G=["D", "F"])),
]
print("DAG 驗證：")
for name, dag in cases:
    probs, _ = check(dag)
    print("  {:<16} -> {}".format(name, "；".join(probs) if probs else "PASS"))
print()

_, order = check(DAG)
level, anc = {}, {}
for n in order:
    level[n] = 1 + max((level[p] for p in DAG[n]), default=0)
    anc[n] = set(DAG[n]).union(*[anc[p] for p in DAG[n]])
by_level = Counter(level.values())
width = max(by_level.values())
groups = [sorted(n for n in DAG if level[n] == L) for L in sorted(by_level) if by_level[L] > 1]
pairs = [(u, v) for i, u in enumerate(order) for v in order[i + 1:] if u not in anc[v] and v not in anc[u]]
fin, prev = {}, {}
for n in order:
    p = max(DAG[n], key=lambda x: fin[x], default=None)
    fin[n] = (fin[p] if p else 0.0) + EST[n]
    prev[n] = p
end = max(fin, key=fin.get)
length = fin[end]
path = []
while end:
    path.append(end)
    end = prev[end]
print("拓撲順序：", " ".join(order))
print("層級：", "  ".join("{}={}".format(n, level[n]) for n in order))
print("最大並行寬度：", width, "；同層可並行的組：", groups)
print("互不依賴（可並行）的節點對：", " ".join("({},{})".format(u, v) for u, v in pairs))
print("關鍵路徑：", "→".join(reversed(path)), "，長度 {:.0f} 分鐘".format(length))
print()

def simulate(n_exec, dur):
    done, log, running = {}, {}, []
    free = list(range(n_exec))
    pending = set(DAG)
    t = 0.0
    while pending or running:
        ready = sorted(n for n in pending if all(p in done for p in DAG[n]))
        while ready and free:
            n = ready.pop(0)
            e = free.pop(0)
            pending.discard(n)
            running.append((t + dur[n], n, e, t))
        running.sort()
        end_t, n, e, s = running.pop(0)
        t = end_t
        done[n] = end_t
        log[n] = (e, s, end_t)
        free.append(e)
        free.sort()
    return t, log

print("{:>9} | {:>9}".format("executor", "總時間"))
print("-" * 22)
for k in (1, 2, 3):
    t, log = simulate(k, EST)
    print("{:>9} | {:>7.1f} 分".format(k, t))
t, log = simulate(2, EST)
print()
print("2 個 executor 時：C 在 executor{} {:.0f}～{:.0f}，F 在 executor{} {:.0f}～{:.0f}".format(log["C"][0], log["C"][1], log["C"][2], log["F"][0], log["F"][1], log["F"][2]))
```

- **預期看到什麼**（全部確定值，沒有亂數）：
  - DAG 驗證 4 行：`原圖 -> PASS`；`B 多依賴 H（循環） -> 可執行：有循環，卡住的節點 B,C,D,E,F,G,H`；`G 的依賴打錯成 E2 -> 可執行：G 依賴不存在的 E2`；`拿掉 G 對 E 的依賴 -> 完整：終點不只一個（E,H），有中間產出沒人接`。
  - 拓撲順序 `A B C D E F G H`；層級 `A=1 B=2 C=3 D=4 E=5 F=3 G=6 H=7`。
  - 最大並行寬度 `2`，同層可並行的組 `[['C', 'F']]`。
  - 可並行的節點對：`(C,F) (D,F) (E,F)`。
  - 關鍵路徑 `A→B→C→D→E→G→H`，長度 `35` 分鐘。
  - executor 表：1 → `40.0`、2 → `35.0`、3 → `35.0`（3 個不會比 2 個快）。
  - 最後一行：`C 在 executor0 10～15，F 在 executor1 10～15`。
- **是否呼叫模型**：否

## 第 3 步：練習 1 前半——Executor 實際排程，與「可並行 ≠ 實際重疊」的陷阱

- **標題**：實際時間有誤差的排程：2 個 executor 省多少？E 與 F 真的重疊了嗎？
- **目的**：照練習 1 的規格（實際時間 = 預估 × uniform(0.8, 1.3)）跑 1 個與 2 個 executor，看並行省下的時間剛好是「不在關鍵路徑上的 F」；並揭開陷阱：E、F 在圖上可並行，但排程器會讓 F 一就緒就先做，結果 F 和 C 重疊、根本沒和 E 重疊——要等 F 變慢，E/F 才真的重疊，而且關鍵路徑會換人。
- **提示（tip）**：「圖上可以並行」說的是依賴關係；「實際有沒有重疊」要看時間軸。觀察重點問的是後者，所以要照實記錄。
- **要跑的內嵌 Python**：

```python
import random
random.seed(202)
DAG = {"A": [], "B": ["A"], "C": ["B"], "D": ["C"], "E": ["D"], "F": ["B"], "G": ["D", "E", "F"], "H": ["G"]}
EST = {n: 5.0 for n in DAG}
ORDER = ["A", "B", "C", "F", "D", "E", "G", "H"]
dur = {n: EST[n] * random.uniform(0.8, 1.3) for n in sorted(DAG)}

def simulate(n_exec, dur):
    done, log, running = {}, {}, []
    free = list(range(n_exec))
    pending = set(DAG)
    t = 0.0
    while pending or running:
        ready = sorted(n for n in pending if all(p in done for p in DAG[n]))
        while ready and free:
            n = ready.pop(0)
            e = free.pop(0)
            pending.discard(n)
            running.append((t + dur[n], n, e, t))
        running.sort()
        end, n, e, s = running.pop(0)
        t = end
        done[n] = end
        log[n] = (e, s, end)
        free.append(e)
        free.sort()
    return t, log

def critical(dur):
    fin, prev = {}, {}
    for n in ORDER:
        p = max(DAG[n], key=lambda x: fin[x], default=None)
        fin[n] = (fin[p] if p else 0.0) + dur[n]
        prev[n] = p
    end = max(fin, key=fin.get)
    length = fin[end]
    path = []
    while end:
        path.append(end)
        end = prev[end]
    return "→".join(reversed(path)), length

def overlap(log):
    _, se, ee = log["E"]
    _, sf, ef = log["F"]
    return se < ef and sf < ee

t1, _ = simulate(1, dur)
t2, log2 = simulate(2, dur)
print("{:<4} | {:>8} | {:>4} | {:>6} | {:>6} | {:>6}".format("節點", "executor", "預估", "實際", "開始", "結束"))
print("-" * 50)
for n in sorted(log2, key=lambda x: (log2[x][1], x)):
    e, s, en = log2[n]
    print("{:<4} | {:>8} | {:>4.0f} | {:>6.2f} | {:>6.2f} | {:>6.2f}".format(n, e, EST[n], dur[n], s, en))
path, length = critical(dur)
print()
print("1 個 executor 總時間：{:.2f} 分鐘".format(t1))
print("2 個 executor 總時間：{:.2f} 分鐘（快 {:.2f} 倍，省下的約等於 F 的 {:.2f} 分鐘）".format(t2, t1 / t2, dur["F"]))
print("關鍵路徑：{}，長度 {:.2f} 分鐘".format(path, length))
print("E 與 F 在圖上互不依賴（可並行），實際重疊執行：{}".format("是" if overlap(log2) else "否"))
print("  F 在 {:.2f}～{:.2f}，E 在 {:.2f}～{:.2f}".format(log2["F"][1], log2["F"][2], log2["E"][1], log2["E"][2]))
print()
slow = dict(dur, F=20.0)
t3, log3 = simulate(2, slow)
path3, len3 = critical(slow)
print("假設 F（宏觀分析）變慢成 20 分鐘：")
print("  總時間 {:.2f} 分鐘；關鍵路徑 {}；E 與 F 實際重疊：{}".format(t3, path3, "是" if overlap(log3) else "否"))
```

- **預期看到什麼**：
  - 時間軸表 8 列（欄位：節點｜executor｜預估｜實際｜開始｜結束），每個「實際」介於 4.00～6.50；C 與 F 開始時間相同（都等 B 做完），分別在 executor 0 與 1。
  - 1 個 executor 總時間約 36～48 分鐘；2 個 executor 約 31～43 分鐘，加速約 1.10～1.23 倍；兩者相差 ≈ F 的實際時間。
  - 關鍵路徑 `A→B→C→D→E→G→H`，長度 = 2 個 executor 的總時間（數字相同）。
  - `E 與 F … 實際重疊執行：否`（必然：F 最多 6.5 分鐘，一定比 C+D 先做完），下一行顯示 F 的結束時間早於 E 的開始時間。
  - F 改成 20 分鐘：`E 與 F 實際重疊：是`、關鍵路徑變成 `A→B→F→G→H`，總時間約 37～46 分鐘（= A+B+G+H 的實際時間 + 20）。
- **是否呼叫模型**：否

## 第 4 步：練習 1 後半——注入三種異常，比較被動與主動 Monitor

- **標題**：數據、進度、邏輯三種異常：passive Monitor 等人類，active Monitor 自己補救
- **目的**：照練習 1 規格注入三種固定異常，讓學習者看到：Monitor 用規則偵測出三種異常並標對類型；active 模式對數據、進度異常自動補救，總時間大約只有 passive 的一半；但邏輯異常（D 說戶外成長最高、F 說戶外宏觀負面）兩種模式都只能交給人類。結果存成 `triad_result.json`。
- **提示（tip）**：passive 的每個警告都卡 60 分鐘（下班沒人回）；active 能修的只有「規則明確」的兩種，結論互相矛盾要人判斷。
- **模擬假設（寫在腳本註解即可）**：passive 的進度異常＝D 照跑 30 分鐘再等人 60 分鐘；active＝跑到 15 分鐘（預估×3）就砍掉、換 executor 用正常時間重跑（模擬上接在同一個時間槽）；邏輯異常＝G 開工前等人 60 分鐘；基準時間的亂數與第 3 步相同（同一種子、同順序抽）。

## 第 5 步：陷阱——Monitor 授權越大，過度修正越多

- **標題**：授權範圍依風險設定：全被動、全主動、依風險分級、分級＋第二證據

            severe += 1
        elif not a["high"] and a["u_fix"] >= 0.9:
            low_fail += 1
    return human, over, severe, low_fail

## 第 6 步：介面層——Schema 版本、寬容的接收者，與 YAML 地獄

- **標題**：訊息信封的演化：嚴格接收者會把新版本擋掉，寬容接收者忽略不認識的欄位；但欄位太多就變 YAML 地獄
- **目的**：讓學習者看到 Schema 版本號＋寬容接收者（不認識的欄位忽略、缺必要欄位才報錯）讓 sender 升級時 receiver 不必同步升級；再用一個玩具模型說明反方：必要欄位越多，能檢查的錯越多，但每次改版對不上的機率和寫設定的時間也跟著漲，存在一個甜蜜點。這一步也是第 7 步 v1 接收器的預演。

def strict_v1(msg):
    miss = [k for k in REQUIRED if k not in msg]
    extra = sorted(k for k in msg if k not in REQUIRED)

print("淨分數最高的欄位數 k =", best[0], "；再往上加欄位就是 YAML 地獄：寫設定比寫程式還久，而且更容易對不上。")

rows.append(("d 可觀測性（input／output 非空）", bool(nodes) and not blank, ("空的：" + ",".join(blank)) if blank else "全部都有"))

flagged = [str(n.get("id")) for n in nodes if sum(str(n.get("desc", "")).count(w) for w in WORDS) >= 2]
rows.append(("e 最小性（啟發式）", not flagged, ("可能不是原子任務：" + ",".join(flagged)) if flagged else "沒有節點被標記"))

crit, width, groups = 0, 0, []
if ok_c:
    level = {}
    for u in order:
        level[u] = 1 + max((level[v] for v in deps[u] if v in deps), default=0)
    cnt = Counter(level.values())
    crit, width = max(level.values()), max(cnt.values())
    groups = [sorted(u for u in level if level[u] == L) for L in sorted(cnt) if cnt[L] > 1]
    note_f = "關鍵路徑 {} 層、最大並行寬度 {}、可並行組 {}".format(crit, width, groups)
else:
    note_f = "有循環或缺依賴，無法計算"
rows.append(("f 並行度（不是一直線）", width >= 2, note_f))

joins = [u for u in deps if len([v for v in deps[u] if v in deps]) >= 2]
lines = ["Monitor 可盯：{} 個節點都有 output 可做數據檢查；多路匯合的節點 {} 是做邏輯一致性檢查的位置。".format(len(nodes) - len(blank), ",".join(joins) if joins else "（無）"),
         "進度：關鍵路徑 {} 層、最大並行寬度 {}，每個節點都可以設「預估 × 3」的進度門檻。".format(crit, width)]
if flagged:
    lines.append("黑盒子風險：{} 的 desc 塞了多個動作，像封裝太深的複合任務，出錯時難定位。".format(",".join(flagged)))
elif nodes:
    longest = max(nodes, key=lambda n: len(str(n.get("desc", ""))))
    lines.append("黑盒子風險：沒有節點被標成非原子；desc 最長的 {} 最可能藏多個步驟，值得再拆。".format(longest.get("id")))
else:
    lines.append("黑盒子風險：沒有節點可評估，整份計畫本身就是黑盒。")

print("資料來源：", source)
print("節點：")
for n in nodes:
    print("  {:<4} 依賴 {:<12} {}".format(str(n.get("id")), ",".join(deps.get(str(n.get("id")), [])) or "-", str(n.get("desc", ""))[:36]))
print()
print("{:<30} | {:<4} | {}".format("檢查", "結果", "說明"))
print("-" * 90)
for name, ok, note in rows:
    print("{:<30} | {:<4} | {}".format(name, "PASS" if ok else "FAIL", note))
print()
for s in lines:
    print("  " + s)

fence = chr(96) * 3
report = "# 第 2 章練習 2：Planner DAG 檢查報告\n\n資料來源：" + source + "\n\n## 模型原始訊息\n\n" + fence + "json\n" + json.dumps(msg, ensure_ascii=False, indent=2) + "\n" + fence + "\n\n## 檢查結果\n\n| 檢查 | 結果 | 說明 |\n|---|---|---|\n"
report += "".join("| {} | {} | {} |\n".format(name, "PASS" if ok else "FAIL", note) for name, ok, note in rows)
report += "\n## 三行結論\n\n" + "\n".join("- " + s for s in lines) + "\n"
(out / "planner_report.md").write_text(report, encoding="utf-8")
print()
print("已寫入", out / "planner_report.md")
